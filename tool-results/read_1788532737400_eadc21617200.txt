   150→function compactINR(v: number): string {
   151→  const abs = Math.abs(v);
   152→  if (abs >= 10000000) return `₹${(v / 10000000).toFixed(1)}Cr`;
   153→  if (abs >= 100000) return `₹${(v / 100000).toFixed(1)}L`;
   154→  if (abs >= 1000) return `₹${(v / 1000).toFixed(0)}K`;
   155→  return `₹${v.toFixed(0)}`;
   156→}

   189→

   198→          () => [] as LowStockMini[]

   201→          firmId: activeFirmId,
   202→          period: currentPeriod(),

   205→          .catch(() => null),

   207→          totals: { openInvoices: number; outstanding: number; overdueInvoices: number; overdue: number };

   213→          rows: Array<{ isOverdue: boolean; overdueDays: number; vendorName: string; dueDate: string }>;
   214→        }>("/api/v1/ledger/aging-pos", { firmId: activeFirmId }).catch(() => null),
   215→      ]);
   216→      setData(dash);
   217→      setLowStock(Array.isArray(low) ? low : []);
   218→      setGstPulse(gst);
   219→
   220→      if (aging) {
   221→        const overdueRows = aging.rows.filter((r) => r.isOverdue);
   222→        setOverduePulse({
   223→          openInvoices: aging.totals.openInvoices,
   224→          outstanding: aging.totals.outstanding,
   225→          overdueInvoices: aging.totals.overdueInvoices,
   226→          overdue: aging.totals.overdue,
   227→          worstOverdueDays: overdueRows.reduce((m, r) => Math.max(m, r.overdueDays), 0),
   228→          topOverdueParty:
   229→            overdueRows.length > 0
   230→              ? [...overdueRows].sort((a, b) => b.overdueDays - a.overdueDays)[0]?.partyName ?? null
   231→              : null,
   232→          nextDueDate:
   233→            aging.rows.length > 0 ? aging.rows.map((r) => r.dueDate).sort()[0] ?? null : null,
   234→          unappliedReceipts: aging.reconciliation?.unappliedReceipts ?? 0,
   235→        });
   236→      } else {
   237→        setOverduePulse(null);
   238→      }
   239→
   240→      // AP pulse — mirror of the AR pulse from the PO-wise aging
   241→      if (ap) {
   242→        const apOverdueRows = ap.rows.filter((r) => r.isOverdue);
   243→        const upcoming = ap.rows
   244→          .map((r) => r.dueDate)
   245→          .filter(Boolean)
   246→          .sort();
   247→        setApPulse({
   248→          openPOs: ap.totals.openPOs,
   249→          outstanding: ap.totals.outstanding,
   250→          overduePOs: ap.totals.overduePOs,
   251→          overdue: ap.totals.overdue,
   252→          worstOverdueDays: apOverdueRows.reduce((m, r) => Math.max(m, r.overdueDays), 0),
   253→          topOverdueVendor:
   254→            apOverdueRows.length > 0
   255→              ? [...apOverdueRows].sort((a, b) => b.overdueDays - a.overdueDays)[0]?.vendorName ?? null
   256→              : null,
   257→          nextDueDate: upcoming[0] ?? null,
   258→        });
   259→      } else {
   260→        setApPulse(null);
   261→      }
   262→    } catch (e) {
   263→      setError(e instanceof Error ? e.message : "Failed to load dashboard");
   264→    } finally {
   265→      setLoading(false);
   266→    }
   267→  }, [activeFirmId]);
   268→
   269→  React.useEffect(() => {
   270→    void load();
   271→  }, [load, refreshKey]);
   272→
   273→  const trendData = React.useMemo(

   275→      (data?.salesTrend ?? []).map((p) => ({
   276→        label: new Date(`${p.date}T00:00:00`).toLocaleDateString("en-IN", {
   277→          day: "2-digit",
   278→          month: "short",
   279→        }),
   280→        total: p.total,
   281→      })),
   282→    [data]
   283→  );
   284→
   285→  const agingRows = React.useMemo(() => {
   286→    const a = data?.arAging;
   287→    if (!a) return [];

   289→      { bucket: "0-30", amount: a.d0_30 },
   290→      { bucket: "31-60", amount: a.d31_60 },
   291→      { bucket: "61-90", amount: a.d61_90 },
   292→      { bucket: "90+", amount: a.d90plus },
   293→    ];
   294→  }, [data]);
   295→
   296→  // Sparkline series for the sales KPI cards
   297→  const trendSpark = React.useMemo(
   298→    () => (data?.salesTrend ?? []).map((p) => p.total),
   299→    [data]
   300→  );
   301→  const monthSpark = React.useMemo(
   302→    () => (data?.monthTrend ?? []).map((p) => p.total),
   303→    [data]
   304→  );
   305→
   306→  const lowStockRail = lowStock.slice(0, 5);
   307→
   308→  return (
   309→    <div className="space-y-4">
   310→      <PageHeader
   311→        title="Dashboard"
   312→        subtitle={`${activeFirm?.firmName ?? data?.firmName ?? "—"} · Executive cockpit`}
   313→        actions={
   314→          <Button
   315→            variant="outline"
   316→            size="sm"
   317→            onClick={() => setRefreshKey((k) => k + 1)}
   318→            disabled={loading}
   319→            className="h-9 gap-2 border-dmk-border-subtle bg-dmk-input-well text-[12.5px] hover:bg-dmk-hover"
   320→          >
   321→            <RefreshCw className={cn("h-3.5 w-3.5", loading && "animate-spin")} />
   322→            Refresh
   323→          </Button>
   324→        }
   325→      />
   326→
   327→      {error && <ErrorText>{error}</ErrorText>}
   328→
   329→      {!data && loading ? (
   330→        <div className="space-y-4">
   331→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   332→            {Array.from({ length: 6 }).map((_, i) => (
   333→              <Skeleton key={i} className="h-[92px] rounded-[10px] bg-dmk-input-well" />
   334→            ))}
   335→          </div>
   336→          <LoadingRows rows={5} />
   337→        </div>
   338→      ) : data ? (
   339→        <>
   340→          {/* ── KPI row (click to drill down) ───────────── */}
   341→          <div className="dmk-enter-stagger grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   342→            <KpiCard label="Today's Sales" value={formatINR(data.todaySales)} tone="orange" icon={IndianRupee}
   343→              spark={trendSpark} sparkColor="#FF6B00"
   344→              onClick={() => setView("sales/invoices")} drillHint="Invoice Register" />
   345→            <KpiCard label="Month Sales" value={formatINR(data.monthSales)} tone="blue" icon={CalendarDays}
   346→              spark={monthSpark} sparkColor="#2563EB"
   347→              onClick={() => setView("reports")} drillHint="Sales Report" />
   348→            <KpiCard
   349→              label="Receivables"
   350→              value={formatINR(data.receivables)}
   351→              tone="orange"
   352→              sub={data.receivablesAdvances > 0.009 ? `Net · incl. ${formatINR(data.receivablesAdvances)} advances` : "Dr — owed by customers"}
   353→              icon={HandCoins}
   354→              onClick={() => setView("finance/aging")}
   355→              drillHint="AR Aging"
   356→            />
   357→            <KpiCard
   358→              label="Payables"
   359→              value={formatINR(data.payables)}
   360→              tone="info"
   361→              sub={data.payablesCredits > 0.009 ? `Net · incl. ${formatINR(data.payablesCredits)} credits` : "Cr — owed to vendors"}
   362→              icon={Banknote}
   363→              onClick={() => setView("finance/aging")}
   364→              drillHint="AP Aging"
   365→            />
   366→            <KpiCard
   367→              label="Cash + Bank"
   368→              value={formatINR(data.cash + data.bank)}
   369→              tone="success"
   370→              sub={`Cash ${formatINR(data.cash)} · Bank ${formatINR(data.bank)}`}
   371→              icon={Wallet}
   372→              onClick={() => setView("finance/daybook")}
   373→              drillHint="Day Book"
   374→            />
   375→            <KpiCard
   376→              label="Inventory Value"
   377→              value={formatINR(data.inventoryValue)}
   378→              tone="default"
   379→              sub={`Damaged: ${formatINR(data.damagedValue)}`}
   380→              icon={Boxes}
   381→              onClick={() => setView("inventory/stock")}

   397→          {apPulse && (

   400→              onDrill={() => {
   401→                requestAgingTab("po");
   402→                setView("finance/aging");
   403→              }}
   404→            />
   405→          )}
   406→
   407→          {/* ── Trend + AR aging ────────────────────────── */}
   408→          <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   409→            <div className="dmk-card p-4 lg:col-span-2">
   410→              <div className="flex items-center justify-between mb-3">
   411→                <div>
   412→                  <h2 className="text-[15px] font-semibold text-dmk-text-primary">Sales Trend</h2>
   413→                  <p className="text-[11px] text-dmk-text-muted">Last 7 days · posted invoices</p>
   414→                </div>
   415→                <Badge tone="info">7 DAYS</Badge>
   416→              </div>
   417→              <div className="h-[260px]">
   418→                <ResponsiveContainer width="100%" height="100%">
   419→                  <AreaChart data={trendData} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   420→                    <defs>
   421→                      <linearGradient id="dmkSalesGradient" x1="0" y1="0" x2="0" y2="1">

   423→                        <stop offset="100%" stopColor="rgba(37,99,235,0)" />

   426→                    <CartesianGrid stroke="#1E2D4A" strokeDasharray="3 3" vertical={false} />

   439→                      tickFormatter={(v) => compactINR(Number(v))}

   441→                    <Tooltip

   443→                      labelStyle={{ color: "var(--text-secondary)", marginBottom: 4 }}
   444→                      itemStyle={{ color: "var(--text-primary)", fontFamily: "var(--font-jetbrains)" }}
   445→                      formatter={(value) => formatINR(Number(value))}
   446→                      cursor={{ stroke: "#2A3F66", strokeDasharray: "3 3" }}
   447→                    />
   448→                    <Area
   449→                      type="monotone"
   450→                      dataKey="total"
   451→                      name="Sales"
   452→                      stroke="#2563EB"
   453→                      strokeWidth={2}

   456→                    />
   457→                  </AreaChart>
   458→                </ResponsiveContainer>
   459→              </div>
   460→            </div>
   461→
   462→            <div className="dmk-card p-4">
   463→              <div className="flex items-center justify-between mb-3">
   464→                <div>
   465→                  <h2 className="text-[15px] font-semibold text-dmk-text-primary">AR Aging</h2>
   466→                  <p className="text-[11px] text-dmk-text-muted">
   467→                    Total outstanding {formatINR(data.arAging.total)}
   468→                  </p>
   469→                </div>
   470→                <Badge tone="dr">Dr</Badge>
   471→              </div>
   472→              <div className="h-[260px]">
   473→                <ResponsiveContainer width="100%" height="100%">
   474→                  <BarChart data={agingRows} layout="vertical" margin={{ top: 4, right: 16, left: 0, bottom: 0 }}>

   477→                      type="number"

   483→                    <YAxis

   493→                      labelStyle={{ color: "var(--text-secondary)", marginBottom: 4 }}

   509→          {/* ── Top products + low-stock rail ───────────── */}
   510→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   511→            <div className="dmk-card p-4">
   512→              <div className="flex items-center justify-between mb-3">
   513→                <div>
   514→                  <h2 className="text-[15px] font-semibold text-dmk-text-primary">Top Products</h2>
   515→                  <p className="text-[11px] text-dmk-text-muted">Last 30 days · by quantity sold</p>

   517→                <ShoppingBag className="h-4 w-4 text-dmk-text-muted" />
