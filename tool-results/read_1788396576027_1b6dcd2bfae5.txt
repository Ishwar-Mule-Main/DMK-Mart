    78→function round2(n: number): number {
    79→  return Math.round((n + Number.EPSILON) * 100) / 100;

   122→/** Client replica of the server estimate (tier → bulk → GST). */
   123→function estimateLine(
   124→  product: Product,
   125→  qty: number,
   126→  manualPct: number | null,
   127→  tierKey: string,
   128→  intra: boolean
   129→) {
   130→  const tierPrice = (product[tierKey as keyof Product] as number) || 0;
   131→  const base = tierPrice > 0 ? tierPrice : product.tier4Retailer;
   132→  const bulk = calculateBulkPricing(base, qty, manualPct ?? 0);
   133→  const rate = product.gstRate;
   134→  const tax = intra
   135→    ? round2(bulk.taxable * (rate / 200)) * 2
   136→    : round2(bulk.taxable * (rate / 100));
   137→  return { bulk, tax: round2(tax), total: round2(bulk.taxable + tax) };
   138→}

   291→    if (!name.trim()) return "Template name is required.";

   340→      setRefresh((r) => r + 1);

   342→      setFormError(e instanceof ApiError ? e.message : "Could not save the template.");

   344→      setSaving(false);

   406→          });
   407→        } else {
   408→          toast({
   409→            variant: "destructive",
   410→            title: "Generation failed",
   411→            description: run?.error ?? "No invoice was generated for this template.",
   412→          });
   413→        }
   414→        setRefresh((r) => r + 1);
   415→      } else {
   416→        setGenResult(res);
   417→        setRefresh((r) => r + 1);
   418→      }

   421→        variant: "destructive",
   422→        title: "Generation failed",
   423→        description: e instanceof ApiError ? e.message : "Could not run recurring generation.",
