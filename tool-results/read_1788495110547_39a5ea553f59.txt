    79→function tierLabel(key: string): string {
    80→  return TIERS.find((t) => t.key === key)?.label ?? "Retailer";
    81→}
    82→
    83→function tierPriceOf(p: Product, tierKey: string): number {
    84→  const map = p as unknown as Record<string, number>;
    85→  return Number(map[tierKey] ?? p.tier4Retailer) || 0;
    86→}
    87→
    88→function tierBadgeTone(key: string): "info" | "success" | "warning" | "neutral" | "dr" {
    89→  switch (key) {
    90→    case "tier1Distributor": return "info";
    91→    case "tier2Wholesale": return "success";
    92→    case "tier3SemiWholesale": return "warning";
    93→    case "tier4Retailer": return "neutral";
    94→    default: return "dr";
    95→  }
    96→}

    98→/** Client-side GST split preview — mirrors lib/gst calculateGST. */
    99→function previewGst(taxable: number, rate: number, seller: string, buyer: string) {
   100→  const intra = seller && buyer && seller === buyer;
   101→  if (intra) return { cgst: round2((taxable * rate) / 200), sgst: round2((taxable * rate) / 200), igst: 0 };
   102→  return { cgst: 0, sgst: 0, igst: round2((taxable * rate) / 100) };
   103→}

   197→  function addProduct(p: Product) {
   198→    setLines((prev) => {
   199→      const existing = prev.find((l) => l.product.id === p.id);
   200→      if (existing) {
   201→        return prev.map((l) => (l.product.id === p.id ? { ...l, qty: l.qty + 1 } : l));
   202→      }
   203→      return [...prev, { product: p, qty: 1, manualDiscPct: 0 }];
   204→    });
   205→    setProdQuery("");
   206→    setProdResults([]);
   207→    setProdOpen(false);
   208→  }
   209→
   210→  function setQty(productId: string, qty: number) {
   211→    setLines((prev) => prev.map((l) => (l.product.id === productId ? { ...l, qty: Math.max(1, qty) } : l)));
   212→  }
   213→  function setDisc(productId: string, pct: number) {
   214→    setLines((prev) => prev.map((l) => (l.product.id === productId ? { ...l, manualDiscPct: Math.min(100, Math.max(0, pct)) } : l)));
   215→  }
   216→  function removeLine(productId: string) {
   217→    setLines((prev) => prev.filter((l) => l.product.id !== productId));
   218→  }

   239→      savings += bp.savings;
   240→      const g = previewGst(bp.taxable, l.product.gstRate, sellerState, buyerState);

   242→      sgst += g.sgst;

   252→    const grand = Math.round(exact);
   253→    const roundOff = round2(grand - exact);
   254→    return { baseSubtotal, taxable, savings, cgst, sgst, igst, exact, grand, roundOff, intra: sellerState === buyerState && !!buyerState };

   264→    }
   265→    return null;
   266→  }, [customer, paymentMode, totals.grand]);

   292→        variant: "destructive",
   293→        title: code === "ERR_CUSTOMER_CREDIT_LOCK" ? "Credit limit blocked" : code === "ERR_INSUFFICIENT_SELLABLE_STOCK" ? "Stock unavailable" : "Sale failed",
   294→        description: `${msg} (${code})`,
   295→      });
   296→    } finally {
   297→      setSubmitting(false);
   298→    }
   299→  }

   312→        icon={ReceiptText}
   313→        actions={
   314→          <Button variant="outline" size="sm" onClick={resetCart} disabled={lines.length === 0} className="h-9 border-dmk-border-subtle text-dmk-text-secondary hover:bg-dmk-hover">
   315→            <Trash2 className="h-4 w-4" /> Clear cart
   316→          </Button>
   317→        }
   318→      />
   319→
   320→      <div className="grid grid-cols-1 lg:grid-cols-[1fr_380px] gap-4 items-start">

   322→        <div className="space-y-4 min-w-0">
   323→          {/* Customer picker */}

   325→            <div className="flex items-center justify-between gap-2">
   326→              <span className="text-[11px] uppercase tracking-wider font-semibold text-dmk-text-muted">Customer</span>
   327→              <Button size="sm" variant="outline" className="h-8 border-dmk-border-subtle text-dmk-text-secondary hover:bg-dmk-hover" onClick={() => setNewCustOpen(true)}>
   328→                <UserPlus className="h-3.5 w-3.5" /> New Customer
   329→              </Button>
   330→            </div>
   331→
   332→            {customer ? (
   333→              <div className="dmk-well p-3 flex flex-wrap items-center gap-x-4 gap-y-2">

   336→                    <span className="text-[14px] font-semibold text-dmk-text-primary truncate">{customer.partyName}</span>
   337→                    <Badge tone={tierBadgeTone(customer.assignedTier)}>{tierLabel(customer.assignedTier)}</Badge>

   343→                  </div>
   344→                </div>
   345→                <div className="flex items-center gap-4 text-right">
   346→                  <div>
   347→                    <p className="text-[10px] uppercase tracking-wider text-dmk-text-muted">Credit limit</p>
   348→                    <p className="font-money text-[13px] text-dmk-text-primary">{formatINR(customer.creditLimit)}</p>

   350→                  <div>
   351→                    <p className="text-[10px] uppercase tracking-wider text-dmk-text-muted">Outstanding</p>

   354→                    </p>

   360→                  >
   361→                    <X className="h-4 w-4" />
   362→                  </button>
   363→                </div>
   364→              </div>
   365→            ) : (

   426→                  {prodResults.map((p, i) => {
   427→                    const tp = tierPriceOf(p, tierKey);

   429→                      <button
   430→                        key={p.id}

   436→                          "w-full text-left px-3 py-2.5 transition-colors border-b border-dmk-border-subtle last:border-b-0",
   437→                          i === prodHi ? "bg-dmk-hover" : "hover:bg-dmk-hover"
   438→                        )}
   439→                      >
   440→                        <div className="flex items-center justify-between gap-3">
   441→                          <span className="text-[13px] text-dmk-text-primary truncate">
   442→                            <span className="font-money text-dmk-text-secondary">{p.sku}</span> · {p.name}
   443→                          </span>
   444→                          <span className="font-money text-[12.5px] text-dmk-gold shrink-0">{formatINR(tp)}</span>
   445→                        </div>
   446→                        <div className="text-[11px] text-dmk-text-muted mt-0.5">

   457→          {/* Cart table */}
   458→          <div className="dmk-card overflow-hidden">
   459→            <div className="overflow-x-auto">
   460→              {lines.length === 0 ? (
   461→                <EmptyState
   462→                  icon={ShoppingCart}
   463→                  title="Cart is empty"
   464→                  hint="Search a product above (SKU or name) and press Enter to add the first match. Quantity unlocks automatic bulk discounts."
   465→                />
   466→              ) : (
   467→                <table className="dmk-table min-w-[860px]">
   468→                  <thead>

   704→// Quick new-customer dialog (B2B)
   705→// ═══════════════════════════════════════════════════════════════
   706→function NewCustomerDialog({
   707→  open,
   708→  onOpenChange,
   709→  onCreated,
   710→}: {
   711→  open: boolean;
   712→  onOpenChange: (o: boolean) => void;
   713→  onCreated: (c: Customer) => void;
   714→}) {
   715→  const { toast } = useToast();
   716→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   717→  const firm = useActiveFirm();
   718→  const [saving, setSaving] = React.useState(false);
   719→  const [form, setForm] = React.useState({
   720→    city: "",
   721→    firmName: "",
   722→    gstin: "",
   723→    stateCode: "27",
   724→    assignedTier: "tier3SemiWholesale",
   725→    creditLimit: "0",
   726→    creditDays: "30",
   727→    phone: "",
   728→  });
   729→
   730→  React.useEffect(() => {
   731→    if (open && firm) setForm((f) => ({ ...f, stateCode: firm.stateCode }));
   732→  }, [open, firm]);
   733→
   734→  const set = (k: keyof typeof form) => (e: React.ChangeEvent<HTMLInputElement>) => setForm((f) => ({ ...f, [k]: e.target.value }));
   735→
   736→  async function submit() {
   737→    if (!activeFirmId) return;
   738→    if (!form.city.trim() || !form.firmName.trim()) {
   739→      toast({ variant: "destructive", title: "Missing fields", description: "City and firm name are required (party name is composed as “City FirmName”)." });
   740→      return;
   741→    }
   742→    setSaving(true);
   743→    try {
   744→      const c = await apiPost<Customer>("/api/v1/customers", {
   745→        firmId: activeFirmId,
   746→        customerType: "B2B",
   747→        city: form.city.trim(),
   748→        firmName: form.firmName.trim(),
   749→        gstin: form.gstin.trim(),
   750→        stateCode: form.stateCode,
   751→        assignedTier: form.assignedTier,
   752→        creditLimit: Number(form.creditLimit) || 0,
   753→        creditDays: Number(form.creditDays) || 0,
   754→        phone: form.phone.trim(),
   755→      });
   756→      onCreated(c);
   757→      setForm((f) => ({ ...f, city: "", firmName: "", gstin: "", phone: "", creditLimit: "0" }));
   758→    } catch (e) {
   759→      const msg = e instanceof ApiError ? e.message : "Could not create customer.";
   760→      toast({ variant: "destructive", title: "Create failed", description: msg });
   761→    } finally {
   762→      setSaving(false);
   763→    }
   764→  }
   765→
   766→  return (
   767→    <Dialog open={open} onOpenChange={onOpenChange}>
   768→      <DialogContent className="dmk-elevated border-dmk-border-medium">
   769→        <DialogHeader>
   770→          <DialogTitle className="text-dmk-text-primary">New B2B customer</DialogTitle>
   771→          <DialogDescription className="text-dmk-text-muted">Location-first naming — party name becomes “City FirmName”.</DialogDescription>
   772→        </DialogHeader>
   773→        <div className="grid grid-cols-2 gap-3">
   774→          <Field label="City *">
   775→            <Input value={form.city} onChange={set("city")} placeholder="e.g. Latur" className={inputCls} />
   776→          </Field>
   777→          <Field label="Firm name *">
   778→            <Input value={form.firmName} onChange={set("firmName")} placeholder="e.g. Ishwar Mule" className={inputCls} />
   779→          </Field>
   780→          <Field label="GSTIN">
   781→            <Input value={form.gstin} onChange={set("gstin")} placeholder="27…" className={cn(inputCls, "font-money")} />
   782→          </Field>
   783→          <Field label="Phone">
   784→            <Input value={form.phone} onChange={set("phone")} placeholder="98…" className={cn(inputCls, "font-money")} />
   785→          </Field>
   786→          <Field label="State code">
   787→            <Select value={form.stateCode} onValueChange={(v) => setForm((f) => ({ ...f, stateCode: v }))}>
   788→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   789→              <SelectContent className="max-h-64">
   790→                {STATES.map((s) => (
   791→                  <SelectItem key={s.code} value={s.code}>{s.name}</SelectItem>
   792→                ))}
   793→              </SelectContent>
   794→            </Select>
   795→          </Field>
   796→          <Field label="Assigned tier">
   797→            <Select value={form.assignedTier} onValueChange={(v) => setForm((f) => ({ ...f, assignedTier: v }))}>
   798→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   799→              <SelectContent>
   800→                {TIERS.slice(0, 4).map((t) => (
   801→                  <SelectItem key={t.key} value={t.key}>{t.label}</SelectItem>
   802→                ))}
   803→              </SelectContent>
   804→            </Select>
   805→          </Field>
   806→          <Field label="Credit limit ₹">
   807→            <Input type="number" min={0} value={form.creditLimit} onChange={set("creditLimit")} className={cn(inputCls, "font-money")} />
   808→          </Field>
   809→          <Field label="Credit days">
   810→            <Input type="number" min={0} value={form.creditDays} onChange={set("creditDays")} className={cn(inputCls, "font-money")} />
   811→          </Field>
   812→        </div>
   813→        <DialogFooter>
   814→          <Button variant="outline" onClick={() => onOpenChange(false)} className="border-dmk-border-medium text-dmk-text-secondary hover:bg-dmk-hover">
   815→            Cancel
   816→          </Button>
   817→          <Button onClick={submit} disabled={saving} className="bg-dmk-yellow text-white hover:bg-dmk-yellow/90">
   818→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Plus className="h-4 w-4" />} Create customer
   819→          </Button>
   820→        </DialogFooter>
   821→      </DialogContent>
   822→    </Dialog>
   823→  );
   824→}
