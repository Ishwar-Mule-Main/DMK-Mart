    97→function defectTone(d: string): "danger" | "warning" | "info" | "neutral" {
    98→  switch (d) {
    99→    case "Broken": return "danger";
   100→    case "Damaged": return "warning";
   101→    case "Defective": return "info";
   102→    default: return "neutral";

   106→function previewGst(taxable: number, rate: number, seller: string, buyer: string) {
   107→  const intra = seller && buyer && seller === buyer;
   108→  if (intra) return { cgst: round2((taxable * rate) / 200), sgst: round2((taxable * rate) / 200), igst: 0 };
   109→  return { cgst: 0, sgst: 0, igst: round2((taxable * rate) / 100) };
   110→}
