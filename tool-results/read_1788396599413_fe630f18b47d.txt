   128→// ── helpers ──────────────────────────────────────────────────────
   129→function round2(n: number): number {
   130→  return Math.round((n + Number.EPSILON) * 100) / 100;
   131→}
   132→
   133→function num(s: string | number): number {
   134→  const n = typeof s === "number" ? s : Number(s);
   135→  return Number.isFinite(n) ? n : 0;
   136→}
   137→
   138→/** Client-side replica of lib/gst calculateGST (server stays source of truth). */
   139→function gstPreview(taxable: number, gstRate: number, intra: boolean) {
   140→  if (intra) {
   141→    return { cgst: round2(taxable * (gstRate / 200)), sgst: round2(taxable * (gstRate / 200)), igst: 0 };
   142→  }
   143→  return { cgst: 0, sgst: 0, igst: round2(taxable * (gstRate / 100)) };
   144→}

   146→function normalizeProducts(res: unknown): Product[] {
   147→  if (Array.isArray(res)) return res as Product[];
   148→  const obj = res as { products?: Product[] } | null;
   149→  return obj?.products ?? [];
   150→}

   158→      </span>
   159→    );
   160→  }
   161→  return <Badge tone="info">Distributor</Badge>;
   162→}

   457→function PoFormDialog({

   474→  const [vendors, setVendors] = React.useState<Vendor[]>([]);

   483→

   488→    apiGet<Vendor[]>("/api/v1/vendors", { firmId: activeFirmId })
   489→      .then((res) => alive && setVendors(res))
   490→      .catch(() => alive && setVendors([]));
   491→    apiGet<Product[]>("/api/v1/products", { firmId: activeFirmId, activeOnly: !editing })
   492→      .then((res) => alive && setProducts(normalizeProducts(res)))

   504→      setVendorId(editing.vendorId);
   505→      setPoDate(toISODate(editing.poDate));
   506→      setVendorBillNo(editing.vendorBillNo ?? "");

   510→        editing.items.map((it) => ({
   511→          productId: it.productId,
   512→          sku: it.sku,
   513→          name: it.productName,
   514→          gstRate: it.gstRate,
   515→          qty: String(it.quantity),
   516→          cost: String(it.unitCost),
   517→        }))
   518→      );
   519→    } else {
   520→      setVendorId("");
   521→      setPoDate(toISODate(new Date()));
   522→      setVendorBillNo("");
   523→      setVendorBillDate("");
   524→      setNotes("");
   525→      setLines([]);
   526→    }
   527→    setPickQuery("");
   528→  }, [open, editing]);
   529→
   530→  const vendor = vendors.find((v) => v.id === vendorId);
   531→  const isManufacturer = vendor?.vendorType === "MANUFACTURER" && !!vendor.brand;
   532→  const intra = !!vendor && vendor.stateCode === firmStateCode;

   881→}

   884→// GRN (Receive) dialog — the money moment

   886→interface GrnLine {
   887→  itemId: string;
   888→  sku: string;
   889→  name: string;
   890→  ordered: number;
   891→  received: number;
   892→  accepted: number;
   893→  damaged: number;
   894→}
   895→
   896→function GrnDialog({
   897→  po,
   898→  onClose,
   899→  onConfirmed,
   900→}: {
   901→  po: PoRow | null;
   902→  onClose: () => void;
   903→  onConfirmed: () => void;
   904→}) {
   905→  const { toast } = useToast();
   906→  const [lines, setLines] = React.useState<GrnLine[]>([]);
   907→  const [receivedDate, setReceivedDate] = React.useState(toISODate(new Date()));
   908→  const [billNo, setBillNo] = React.useState("");
   909→  const [billDate, setBillDate] = React.useState("");
   910→  const [note, setNote] = React.useState("");
   911→  const [saving, setSaving] = React.useState(false);
   912→
   913→  React.useEffect(() => {
   914→    if (!po) return;
   915→    setLines(
   916→      po.items.map((it) => ({
   917→        itemId: it.id,
   918→        sku: it.sku,
   919→        name: it.productName,
   920→        ordered: it.quantity,
   921→        received: it.quantity,
   922→        accepted: it.quantity,
   923→        damaged: 0,
   924→      }))
   925→    );
   926→    setReceivedDate(toISODate(new Date()));
   927→    setBillNo(po.vendorBillNo ?? "");
   928→    setBillDate(po.vendorBillDate ? toISODate(po.vendorBillDate) : "");
   929→    setNote("");
   930→  }, [po]);
   931→
   932→  function setReceived(i: number, v: number) {
   933→    setLines((ls) =>
   934→      ls.map((l, idx) => {
   935→        if (idx !== i) return l;
   936→        const received = Math.max(0, round2(v));
   937→        const accepted = Math.min(l.accepted, received);
   938→        return { ...l, received, accepted, damaged: round2(received - accepted) };
   939→      })
   940→    );
   941→  }
   942→
   943→  function setAccepted(i: number, v: number) {
   944→    setLines((ls) =>
   945→      ls.map((l, idx) => {
   946→        if (idx !== i) return l;
   947→        const accepted = Math.min(Math.max(0, round2(v)), l.received);
   948→        return { ...l, accepted, damaged: round2(l.received - accepted) };
   949→      })
   950→    );
   951→  }
   952→
   953→  function setDamaged(i: number, v: number) {
   954→    setLines((ls) =>
   955→      ls.map((l, idx) => {
   956→        if (idx !== i) return l;
   957→        const damaged = Math.min(Math.max(0, round2(v)), round2(l.received - l.accepted));
   958→        return { ...l, damaged };
   959→      })
   960→    );
   961→  }
   962→
   963→  const totalAccepted = round2(lines.reduce((s, l) => s + l.accepted, 0));
   964→  const totalDamaged = round2(lines.reduce((s, l) => s + l.damaged, 0));
   965→  const anyShort = lines.some((l) => l.received < l.ordered);
   966→  const allValid = lines.every(
   967→    (l) => l.accepted >= 0 && l.damaged >= 0 && l.damaged <= round2(l.received - l.accepted) + 0.001
   968→  );
   969→
   970→  async function confirm() {
   971→    if (!po || !allValid) return;
   972→    setSaving(true);
   973→    try {
   974→      await apiPost<{ po: PoRow; journal: unknown }>(`/api/v1/purchase-orders/${po.id}/confirm`, {
   975→        received: lines.map((l) => ({ itemId: l.itemId, acceptedQty: l.accepted, damagedQty: l.damaged })),
   976→        receivedDate,
   977→        note: note.trim(),
   978→        vendorBillNo: billNo.trim(),
   979→        vendorBillDate: billDate || null,
   980→      });
   981→      toast({
   982→        title: `GRN confirmed — ${po.poNumber}`,
   983→        description: billNo.trim()
   984→          ? `Stock updated · Vendor payable increased · Journal posted · Bill ${billNo.trim()} recorded`
   985→          : "Stock updated · Vendor payable increased · Journal posted",
   986→      });
   987→      onConfirmed();
   988→      onClose();
   989→    } catch (e) {
   990→      toast({
   991→        variant: "destructive",
   992→        title: "GRN failed",
   993→        description: e instanceof ApiError ? e.message : "Could not confirm receipt.",
   994→      });
   995→    } finally {
   996→      setSaving(false);
   997→    }
   998→  }
   999→
  1000→  return (
  1001→    <Dialog open={!!po} onOpenChange={(o) => !o && onClose()}>
  1002→      <DialogContent className="sm:max-w-3xl dmk-elevated border-dmk-border-medium max-h-[94vh] overflow-y-auto">
  1003→        <DialogHeader>
  1004→          <DialogTitle className="text-dmk-text-primary flex items-center gap-2">
  1005→            <PackageCheck className="h-5 w-5 text-dmk-success" /> Goods Receipt Note — {po?.poNumber}
  1006→          </DialogTitle>
  1007→          <DialogDescription className="text-dmk-text-muted">
  1008→            {po?.vendor?.vendorName} · Order value <span className="font-money">{formatINR(po?.grandTotal ?? 0)}</span> —
  1009→            Accepted qty → sellable stock · Damaged qty → quarantine pool
  1010→          </DialogDescription>
  1011→        </DialogHeader>
  1012→
  1013→        <div className="dmk-card overflow-hidden">
  1014→          <div className="overflow-x-auto">
  1015→            <table className="dmk-table min-w-[780px]">
  1016→              <thead>
  1017→                <tr>
  1018→                  <th>Product</th>
  1019→                  <th className="text-right">Ordered</th>
  1020→                  <th className="text-right w-[104px]">Received</th>
  1021→                  <th className="text-right w-[104px]">Accepted</th>
  1022→                  <th className="text-right w-[104px]">Damaged</th>
  1023→                  <th />
  1024→                </tr>
