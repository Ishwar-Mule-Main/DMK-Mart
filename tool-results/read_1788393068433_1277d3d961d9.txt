   263→function compactINR(v: number): string {
   264→  const abs = Math.abs(v);
   265→  if (abs >= 10000000) return `₹${(v / 10000000).toFixed(1)}Cr`;
   266→  if (abs >= 100000) return `₹${(v / 100000).toFixed(1)}L`;
   267→  if (abs >= 1000) return `₹${(v / 1000).toFixed(0)}K`;
   268→  return `₹${v.toFixed(0)}`;
   269→}

   798→
   799→function ValuationReport({
   800→  rows,
   801→  totals,
   802→  method,
   803→}: {
   804→  rows: ValuationRow[];
   805→  totals?: { stockValue: number; damagedValue: number; totalValue?: number };
   806→  method?: string;
   807→}) {

   810→  const totalValue = totals?.totalValue ?? stockValue + damagedValue;
   811→  const costValue = rows.reduce((s, r) => s + r.purchaseCost * r.stockQuantity, 0);
   812→  const variance = Math.round((stockValue - costValue) * 100) / 100;

   821→        <div className="dmk-kpi p-4">
   822→          <span className="text-[11px] uppercase tracking-wider font-semibold text-dmk-text-muted">Quarantine @ WAC</span>

   830→          <span className="text-[11px] uppercase tracking-wider font-semibold text-dmk-text-muted">Variance vs Cost</span>
   831→          <span className={cn("font-money text-[19px] font-semibold block mt-2", variance === 0 ? "text-dmk-text-primary" : variance > 0 ? "text-dmk-info" : "text-dmk-warning")}>

   845→      <div className="dmk-card overflow-hidden">

   865→                return (

   867→                    <td className="font-money text-[12px] text-dmk-text-secondary">{r.sku}</td>
