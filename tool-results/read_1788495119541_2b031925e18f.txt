   139→// ── helpers ──────────────────────────────────────────────────────
   140→function round2(n: number): number {
   141→  return Math.round((n + Number.EPSILON) * 100) / 100;
   142→}
   143→
   144→function num(s: string | number): number {
   145→  const n = typeof s === "number" ? s : Number(s);
   146→  return Number.isFinite(n) ? n : 0;
   147→}

   149→/** Client-side replica of lib/gst calculateGST (server stays source of truth). */
   150→function gstPreview(taxable: number, gstRate: number, intra: boolean) {
   151→  if (intra) {
   152→    return { cgst: round2(taxable * (gstRate / 200)), sgst: round2(taxable * (gstRate / 200)), igst: 0 };
   153→  }
   154→  return { cgst: 0, sgst: 0, igst: round2(taxable * (gstRate / 100)) };
   155→}

   157→function normalizeProducts(res: unknown): Product[] {
   158→  if (Array.isArray(res)) return res as Product[];
   159→  const obj = res as { products?: Product[] } | null;
   160→  return obj?.products ?? [];
   161→}
