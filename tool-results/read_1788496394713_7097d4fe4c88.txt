   467→// DATA & BACKUP — firm-level JSON export (R1/R20: firm = account)
   468→// ═══════════════════════════════════════════════════════════════
   469→
   470→function BackupCard({
   471→  firm,
   472→  counts,
   473→}: {
   474→  firm?: Firm;
   475→  counts?: FirmCounts;
   476→}) {
   477→  const { toast } = useToast();
   478→  const [exporting, setExporting] = React.useState(false);
   479→  const [lastExport, setLastExport] = React.useState<string | null>(null);
   480→
   481→  async function exportBackup() {
   482→    if (!firm || exporting) return;
   483→    setExporting(true);
   484→    try {
   485→      const res = await fetch(`/api/v1/backup?firmId=${firm.id}`);
   486→      if (!res.ok) {
   487→        const body = (await res.json().catch(() => null)) as { message?: string } | null;
   488→        throw new Error(body?.message ?? `Export failed (${res.status})`);
   489→      }
   490→      const payload = await res.json();
   491→      const stamp = new Date().toISOString().slice(0, 10);
   492→      const totalRecords = payload?.counts
   493→        ? Object.values(payload.counts as Record<string, number>).reduce((a, b) => a + b, 0)
   494→        : 0;
   495→      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
   496→      const url = URL.createObjectURL(blob);
   497→      const a = document.createElement("a");
   498→      a.href = url;
   499→      a.download = `dmk-backup-${(firm.firmCode || firm.firmName).replace(/\s+/g, "-").toLowerCase()}-${stamp}.json`;
   500→      document.body.appendChild(a);
   501→      a.click();
   502→      a.remove();
   503→      URL.revokeObjectURL(url);
   504→      setLastExport(new Date().toLocaleString("en-IN"));
   505→      toast({
   506→        title: "Backup downloaded",
   507→        description: `${totalRecords} records exported for ${firm.firmName}.`,
   508→      });
   509→    } catch (e) {
   510→      toast({
   511→        variant: "destructive",
   512→        title: "Backup failed",
   513→        description: e instanceof Error ? e.message : "Unknown error",
   514→      });
   515→    } finally {
   516→      setExporting(false);
   517→    }
   518→  }
   519→

   621→/** Accept both the pure envelope and the { ok, data } fetch wrapper our export UI writes to disk. */
   622→function unwrapEnvelopeFile(raw: unknown): Record<string, unknown> {
   623→  const obj = raw && typeof raw === "object" && !Array.isArray(raw) ? (raw as Record<string, unknown>) : {};
   624→  if (typeof obj.format === "string" && obj.format !== "") return obj;
   625→  const inner = obj.data;
   626→  if (inner && typeof inner === "object" && !Array.isArray(inner) && typeof (inner as Record<string, unknown>).format === "string") {
   627→    return inner as Record<string, unknown>;
   628→  }
   629→  return obj;
   630→}
   631→
   632→function buildRestorePreview(fileName: string, raw: unknown): BackupPreview {
   633→  const env = unwrapEnvelopeFile(raw);
   634→  const data = env.data && typeof env.data === "object" && !Array.isArray(env.data) ? (env.data as Record<string, unknown>) : {};
   635→  const firm = env.firm && typeof env.firm === "object" && !Array.isArray(env.firm) ? (env.firm as Record<string, unknown>) : {};
   636→
   637→  const issues: string[] = [];
   638→  const warnings: string[] = [];
   639→  const format = typeof env.format === "string" ? env.format : "";
   640→  const version = typeof env.version === "number" ? env.version : Number(env.version);
   641→  if (format !== "dmk-mart-erp-backup") {
   642→    issues.push(`Not a DMK backup envelope — format is ${format ? `"${format}"` : "missing"}`);
   643→  }
   644→  if (version !== 1 && version !== 2) {
   645→    issues.push(`Unsupported backup version — expected 1 or 2, got ${Number.isFinite(version) ? version : "unknown"}`);
   646→  }
   647→  if (typeof firm.firmName !== "string" || firm.firmName === "") {
   648→    issues.push("Envelope is missing the firm profile (firm.firmName)");
   649→  }
   650→  for (const key of REQUIRED_ENV_KEYS) {
   651→    if (!Array.isArray(data[key])) {
   652→      warnings.push(`data.${key} is missing or not an array — the API will reject this envelope`);
   653→    }
   654→  }
   655→
   656→  const counts: Record<string, number> = {};
   657→  let totalRecords = 0;
   658→  for (const [key] of BACKUP_COUNT_KEYS) {
   659→    const arr = data[key];
   660→    const n = Array.isArray(arr) ? arr.length : 0;
   661→    totalRecords += n;
   662→    if (Array.isArray(arr)) {
   663→      const nested = BACKUP_NESTED_KEYS[key] ?? [];
   664→      for (const row of arr) {
   665→        if (!row || typeof row !== "object") continue;
   666→        for (const nk of nested) {
   667→          const sub = (row as Record<string, unknown>)[nk];
   668→          if (Array.isArray(sub)) {
   669→            // child rows count toward the "rows will be created" total,
   670→            // but stay OUT of the parent tile so labels stay truthful
   671→            totalRecords += sub.length;
   672→          }
   673→        }
   674→      }
   675→    }
   676→    counts[key] = n;
   677→  }
   678→
   679→  return {
   680→    fileName,
   681→    format,
   682→    version: Number.isFinite(version) ? version : 0,
   683→    generatedAt: typeof env.generatedAt === "string" ? env.generatedAt : "",
   684→    firmName: typeof firm.firmName === "string" ? firm.firmName : "",
   685→    firmCode: typeof firm.firmCode === "string" ? firm.firmCode : "",
   686→    counts,
   687→    totalRecords,
   688→    issues,
   689→    warnings,
   690→  };
   691→}
