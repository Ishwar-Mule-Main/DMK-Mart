   454→
   455→function FirmDialog({
   456→  open,
   457→  onOpenChange,
   458→  mode,
   459→  firm,
   460→  onSaved,
   461→}: {
   462→  open: boolean;
   463→  onOpenChange: (v: boolean) => void;
   464→  mode: "create" | "edit";
   465→  firm?: Firm | null;
   466→  onSaved: (newFirmId?: string) => void;
   467→}) {
   468→  const { toast } = useToast();
   469→  const [form, setForm] = React.useState<FirmForm>(EMPTY_FORM);
   470→  const [saving, setSaving] = React.useState(false);
   471→  const [formError, setFormError] = React.useState<string | null>(null);
   472→
   473→  React.useEffect(() => {
   474→    if (!open) return;
   475→    setFormError(null);
   476→    setForm(mode === "edit" && firm ? formFromFirm(firm) : { ...EMPTY_FORM, invoicePrefix: "" });
   477→  }, [open, mode, firm]);
   478→
   479→  function set<K extends keyof FirmForm>(key: K, value: FirmForm[K]) {
   480→    setForm((prev) => ({ ...prev, [key]: value }));
   481→  }
   482→
   483→  const gstinPrefix = /^\d{2}/.test(form.gstin) ? form.gstin.slice(0, 2) : "";
   484→  const createValid = form.firmName.trim() !== "" && form.firmCode.trim() !== "";
   485→  const editValid = form.firmName.trim() !== "";
   486→  const stateName =
   487→    STATES.find((s) => s.code === form.state)?.name.replace(/\s*\(\d+\)$/, "") ??
   488→    (mode === "edit" && firm ? firm.state : "Maharashtra");
   489→
   490→  async function save() {
   491→    setSaving(true);
   492→    setFormError(null);
   493→    try {
   494→      if (mode === "create") {
   495→        const res = await apiPost<{ firm: Firm }>("/api/v1/firms", {
   496→          firmName: form.firmName.trim(),
   497→          firmCode: form.firmCode.trim().toUpperCase(),
   498→          gstin: form.gstin.trim(),
   499→          state: stateName,
   500→          stateCode: form.state,
   501→          address: form.address.trim(),
   502→          phone: form.phone.trim(),
   503→          email: form.email.trim(),
   504→          bankName: form.bankName.trim(),
   505→          bankAccount: form.bankAccount.trim(),
   506→          ifsc: form.ifsc.trim().toUpperCase(),
   507→          financialYear: form.financialYear,
   508→          invoicePrefix: form.invoicePrefix.trim() || undefined,
   509→          openingCash: Number(form.openingCash) || 0,
   510→          openingBank: Number(form.openingBank) || 0,
   511→        });
   512→        onOpenChange(false);
   513→        onSaved(res.firm.id);
   514→      } else if (firm) {
   515→        await apiPatch(`/api/v1/firms/${firm.id}`, {
   516→          firmName: form.firmName.trim(),
   517→          gstin: form.gstin.trim(),
   518→          state: stateName,
   519→          stateCode: form.state,
   520→          address: form.address.trim(),
   521→          phone: form.phone.trim(),
   522→          email: form.email.trim(),
   523→          bankName: form.bankName.trim(),
   524→          bankAccount: form.bankAccount.trim(),
   525→          ifsc: form.ifsc.trim().toUpperCase(),
   526→          financialYear: form.financialYear,
   527→          invoicePrefix: form.invoicePrefix.trim(),
   528→        });
   529→        onOpenChange(false);
   530→        onSaved();
   531→      }
   532→    } catch (e) {
   533→      const msg = e instanceof ApiError ? `${e.message}${e.code ? ` (${e.code})` : ""}` : "Request failed";
   534→      setFormError(msg);
   535→      toast({ variant: "destructive", title: mode === "create" ? "Could not create firm" : "Could not update firm", description: msg });
   536→    } finally {
   537→      setSaving(false);
   538→    }
   539→  }
   540→
   541→  return (
   542→    <Dialog open={open} onOpenChange={onOpenChange}>
   543→      <DialogContent className="max-w-2xl bg-dmk-bg-secondary border-dmk-border-medium max-h-[90vh] overflow-y-auto">

   545→          <DialogTitle className="text-dmk-text-primary">
   546→            {mode === "create" ? "Create New Firm" : `Edit ${firm?.firmName ?? "Firm"}`}
   547→          </DialogTitle>
   548→          <DialogDescription className="text-dmk-text-muted text-[12px]">
   549→            {mode === "create"
   550→              ? "A new firm is a fresh isolated universe — chart of accounts is seeded automatically and the opening journal posts on save."
   551→              : "Profile fields feed invoices, A4 documents and GST place-of-supply logic. Firm code and opening balances are immutable."}
   552→          </DialogDescription>
   553→        </DialogHeader>
   554→
   555→        <div className="space-y-3">
   556→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   557→            <Field label="Firm Name *">
   558→              <Input value={form.firmName} onChange={(e) => set("firmName", e.target.value)} placeholder="DMK Mart" className={inputCls} />
   559→            </Field>
   560→            <Field label="Firm Code *" hint={mode === "edit" ? "Immutable after creation" : "Short unique code, e.g. DMK"}>
   561→              <Input
   562→                value={form.firmCode}
   563→                onChange={(e) => set("firmCode", e.target.value.toUpperCase())}
   564→                placeholder="DMK"
   565→                disabled={mode === "edit"}
   566→                className={cn(inputCls, "font-money uppercase")}
   567→              />
   568→            </Field>
   569→          </div>
   570→
   571→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   572→            <Field label="GSTIN" hint={gstinPrefix ? `First 2 digits “${gstinPrefix}” auto-map to state code` : "15 characters; first 2 digits = state code"}>
   573→              <Input
   574→                value={form.gstin}
   575→                onChange={(e) => set("gstin", e.target.value.toUpperCase())}
   576→                placeholder="27AAAAA0000A1Z5"
   577→                maxLength={15}
   578→                className={cn(inputCls, "font-money uppercase")}
   579→              />
   580→            </Field>
   581→            <Field label="State">
   582→              <Select
   583→                value={form.state}
   584→                onValueChange={(v) => {
   585→                  set("state", v);
   586→                  set("stateCode", v);
   587→                }}
   588→              >
   589→                <SelectTrigger className={cn(inputCls, "w-full")}>
   590→                  <SelectValue placeholder="Select state" />
   591→                </SelectTrigger>
   592→                <SelectContent className="bg-dmk-bg-tertiary border-dmk-border-medium max-h-64">
   593→                  {STATES.map((s) => (
   594→                    <SelectItem key={s.code} value={s.code} className="text-[12.5px]">
   595→                      {s.name}
   596→                    </SelectItem>
   597→                  ))}
   598→                </SelectContent>
   599→              </Select>
   600→            </Field>
   601→          </div>
   602→
   603→          <Field label="Address">
   604→            <Textarea value={form.address} onChange={(e) => set("address", e.target.value)} rows={2} placeholder="Shop no, street, city, PIN" className="bg-dmk-input-well border-dmk-border-subtle text-[13px] text-dmk-text-primary dmk-input resize-none" />
   605→          </Field>
   606→
   607→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   608→            <Field label="Phone">
   609→              <Input value={form.phone} onChange={(e) => set("phone", e.target.value)} placeholder="+91…" className={inputCls} />
   610→            </Field>
   611→            <Field label="Email">
   612→              <Input value={form.email} onChange={(e) => set("email", e.target.value)} placeholder="owner@firm.in" className={inputCls} />
   613→            </Field>
   614→          </div>
   615→
   616→          <div className="dmk-well p-3 space-y-3">
   617→            <p className="text-[11px] font-semibold uppercase tracking-wider text-dmk-text-muted">Bank Details (printed on A4 invoices)</p>
   618→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   619→              <Field label="Bank Name">
   620→                <Input value={form.bankName} onChange={(e) => set("bankName", e.target.value)} placeholder="HDFC Bank" className={inputCls} />
   621→              </Field>

   626→                <Input value={form.ifsc} onChange={(e) => set("ifsc", e.target.value.toUpperCase())} className={cn(inputCls, "font-money uppercase")} />
   627→              </Field>
   628→            </div>
   629→          </div>
   630→
   631→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   632→            <Field label="Financial Year">
   633→              <Select value={form.financialYear} onValueChange={(v) => set("financialYear", v)}>
   634→                <SelectTrigger className={cn(inputCls, "w-full")}>
   635→                  <SelectValue />
   636→                </SelectTrigger>
   637→                <SelectContent className="bg-dmk-bg-tertiary border-dmk-border-medium">

   655→
   656→          {mode === "create" && (

   684→        </div>
   685→
   686→        <DialogFooter>
