   459→   459→function BackupCard({
   460→   460→  firm,
   461→   461→  counts,
   462→   462→}: {
   463→   463→  firm?: Firm;
   464→   464→  counts?: FirmCounts;
   465→   465→}) {
   466→   466→  const { toast } = useToast();
   467→   467→  const [exporting, setExporting] = React.useState(false);
   468→   468→  const [lastExport, setLastExport] = React.useState<string | null>(null);
   469→   469→
   470→   470→  async function exportBackup() {
   471→   471→    if (!firm || exporting) return;
   472→   472→    setExporting(true);
   473→   473→    try {
   474→   474→      const res = await fetch(`/api/v1/backup?firmId=${firm.id}`);
   475→   475→      if (!res.ok) {
   476→   476→        const body = (await res.json().catch(() => null)) as { message?: string } | null;
   477→   477→        throw new Error(body?.message ?? `Export failed (${res.status})`);
   478→   478→      }
   479→   479→      const payload = await res.json();
   480→   480→      const stamp = new Date().toISOString().slice(0, 10);
   481→   481→      const totalRecords = payload?.counts
   482→   482→        ? Object.values(payload.counts as Record<string, number>).reduce((a, b) => a + b, 0)
   483→   483→        : 0;
   484→   484→      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
   485→   485→      const url = URL.createObjectURL(blob);
   486→   486→      const a = document.createElement("a");
   487→   487→      a.href = url;
   488→   488→      a.download = `dmk-backup-${(firm.firmCode || firm.firmName).replace(/\s+/g, "-").toLowerCase()}-${stamp}.json`;
   489→   489→      document.body.appendChild(a);
   490→   490→      a.click();
   491→   491→      a.remove();
   492→   492→      URL.revokeObjectURL(url);
   493→   493→      setLastExport(new Date().toLocaleString("en-IN"));
   494→   494→      toast({
   495→   495→        title: "Backup downloaded",
   496→   496→        description: `${totalRecords} records exported for ${firm.firmName}.`,
   497→   497→      });
   498→   498→    } catch (e) {
   499→   499→      toast({
   500→   500→        variant: "destructive",
   501→   501→        title: "Backup failed",
   502→   502→        description: e instanceof Error ? e.message : "Unknown error",
   503→   503→      });
   504→   504→    } finally {
   505→   505→      setExporting(false);
   506→   506→    }
   507→   507→  }
   508→   508→
   509→   509→  const recordCount = counts
   510→   510→    ? counts.products + counts.customers + counts.vendors + counts.invoices + counts.purchaseOrders
   511→   511→    : null;
   512→   512→
   513→   513→  return (
   514→   514→    <div className="dmk-card p-5">
   515→   515→      <div className="flex items-center gap-2.5 mb-3">
   516→   516→        <DatabaseBackup className="h-4 w-4 text-dmk-info" />
   517→   517→        <h2 className="text-[15px] font-semibold text-dmk-text-primary">Data &amp; Backup</h2>
   518→   518→        {firm && <Badge tone="info">{firm.firmName}</Badge>}
   519→   519→      </div>
   520→   520→      <p className="text-[12.5px] text-dmk-text-secondary mb-3">
   521→   521→        Download the complete data universe of the active firm as a versioned JSON envelope — firm profile, masters,
   522→   522→        documents, subledger allocations, journals, inventory audit trail and GSTR-2B records. Keep it with your
   523→   523→        accountant or feed it into any restore pipeline.
   524→   524→      </p>
   525→   525→      <div className="flex flex-wrap items-center gap-3">
   526→   526→        <Button
   527→   527→          onClick={() => void exportBackup()}
   528→   528→          disabled={!firm || exporting}
   529→   529→          className="h-9 gap-2 bg-dmk-info text-[12.5px] font-semibold text-white hover:bg-dmk-info/85 disabled:opacity-50"
   530→   530→        >
   531→   531→          {exporting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <DatabaseBackup className="h-3.5 w-3.5" />}
   532→   532→          {exporting ? "Preparing…" : "Export full backup (JSON)"}
   533→   533→        </Button>
   534→   534→        <span className="text-[11.5px] text-dmk-text-muted">
   535→   535→          {recordCount !== null
   536→   536→            ? `≈ ${recordCount} core records · ${counts?.invoices ?? 0} invoices · ${counts?.purchaseOrders ?? 0} POs`
   537→   537→            : "Entity counts load with the firm list"}
   538→   538→        </span>
   539→   539→        {lastExport && (
   540→   540→          <span className="dmk-badge bg-dmk-success/10 text-dmk-success">Last export {lastExport}</span>
   541→   541→        )}
   542→   542→      </div>
   543→   543→    </div>
   544→   544→  );
   545→   545→}
