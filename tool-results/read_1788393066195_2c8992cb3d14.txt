   458→
   459→function BackupCard({
   460→  firm,
   461→  counts,
   462→}: {
   463→  firm?: Firm;
   464→  counts?: FirmCounts;
   465→}) {
   466→  const { toast } = useToast();
   467→  const [exporting, setExporting] = React.useState(false);
   468→  const [lastExport, setLastExport] = React.useState<string | null>(null);
   469→
   470→  async function exportBackup() {
   471→    if (!firm || exporting) return;
   472→    setExporting(true);
   473→    try {
   474→      const res = await fetch(`/api/v1/backup?firmId=${firm.id}`);
   475→      if (!res.ok) {
   476→        const body = (await res.json().catch(() => null)) as { message?: string } | null;
   477→        throw new Error(body?.message ?? `Export failed (${res.status})`);
   478→      }
   479→      const payload = await res.json();
   480→      const stamp = new Date().toISOString().slice(0, 10);
   481→      const totalRecords = payload?.counts
   482→        ? Object.values(payload.counts as Record<string, number>).reduce((a, b) => a + b, 0)
   483→        : 0;
   484→      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
   485→      const url = URL.createObjectURL(blob);
   486→      const a = document.createElement("a");
   487→      a.href = url;
   488→      a.download = `dmk-backup-${(firm.firmCode || firm.firmName).replace(/\s+/g, "-").toLowerCase()}-${stamp}.json`;
   489→      document.body.appendChild(a);
   490→      a.click();
   491→      a.remove();
   492→      URL.revokeObjectURL(url);
   493→      setLastExport(new Date().toLocaleString("en-IN"));
   494→      toast({
   495→        title: "Backup downloaded",
   496→        description: `${totalRecords} records exported for ${firm.firmName}.`,
   497→      });
   498→    } catch (e) {
   499→      toast({
   500→        variant: "destructive",
   501→        title: "Backup failed",
   502→        description: e instanceof Error ? e.message : "Unknown error",
   503→      });
   504→    } finally {
   505→      setExporting(false);
   506→    }
   507→  }
   508→
   509→  const recordCount = counts
   510→    ? counts.products + counts.customers + counts.vendors + counts.invoices + counts.purchaseOrders
   511→    : null;
   512→
   513→  return (
   514→    <div className="dmk-card p-5">
   515→      <div className="flex items-center gap-2.5 mb-3">
   516→        <DatabaseBackup className="h-4 w-4 text-dmk-info" />
   517→        <h2 className="text-[15px] font-semibold text-dmk-text-primary">Data &amp; Backup</h2>
   518→        {firm && <Badge tone="info">{firm.firmName}</Badge>}
   519→      </div>
   520→      <p className="text-[12.5px] text-dmk-text-secondary mb-3">
   521→        Download the complete data universe of the active firm as a versioned JSON envelope — firm profile, masters,
   522→        documents, subledger allocations, journals, inventory audit trail and GSTR-2B records. Keep it with your
   523→        accountant or feed it into any restore pipeline.
   524→      </p>
   525→      <div className="flex flex-wrap items-center gap-3">
   526→        <Button
   527→          onClick={() => void exportBackup()}
   528→          disabled={!firm || exporting}
   529→          className="h-9 gap-2 bg-dmk-info text-[12.5px] font-semibold text-white hover:bg-dmk-info/85 disabled:opacity-50"

   531→          {exporting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <DatabaseBackup className="h-3.5 w-3.5" />}
   532→          {exporting ? "Preparing…" : "Export full backup (JSON)"}
   533→        </Button>
   534→        <span className="text-[11.5px] text-dmk-text-muted">
   535→          {recordCount !== null

   548→// FIRM DIALOG (create / edit)
   549→// ═══════════════════════════════════════════════════════════════
   550→
   551→function FirmDialog({
   552→  open,
   553→  onOpenChange,
   554→  mode,
   555→  firm,
   556→  onSaved,
   557→}: {
   558→  open: boolean;
   559→  onOpenChange: (v: boolean) => void;
   560→  mode: "create" | "edit";
   561→  firm?: Firm | null;
   562→  onSaved: (newFirmId?: string) => void;
   563→}) {
   564→  const { toast } = useToast();
   565→  const [form, setForm] = React.useState<FirmForm>(EMPTY_FORM);
   566→  const [saving, setSaving] = React.useState(false);
   567→  const [formError, setFormError] = React.useState<string | null>(null);
   568→
   569→  React.useEffect(() => {
   570→    if (!open) return;
   571→    setFormError(null);
   572→    setForm(mode === "edit" && firm ? formFromFirm(firm) : { ...EMPTY_FORM, invoicePrefix: "" });
   573→  }, [open, mode, firm]);
   574→
   575→  function set<K extends keyof FirmForm>(key: K, value: FirmForm[K]) {
   576→    setForm((prev) => ({ ...prev, [key]: value }));
   577→  }
   578→
   579→  const gstinPrefix = /^\d{2}/.test(form.gstin) ? form.gstin.slice(0, 2) : "";
   580→  const createValid = form.firmName.trim() !== "" && form.firmCode.trim() !== "";
   581→  const editValid = form.firmName.trim() !== "";
   582→  const stateName =
   583→    STATES.find((s) => s.code === form.state)?.name.replace(/\s*\(\d+\)$/, "") ??
   584→    (mode === "edit" && firm ? firm.state : "Maharashtra");
   585→
   586→  async function save() {
   587→    setSaving(true);
   588→    setFormError(null);
   589→    try {
   590→      if (mode === "create") {
   591→        const res = await apiPost<{ firm: Firm }>("/api/v1/firms", {
   592→          firmName: form.firmName.trim(),
   593→          firmCode: form.firmCode.trim().toUpperCase(),
   594→          gstin: form.gstin.trim(),
   595→          state: stateName,
   596→          stateCode: form.state,
   597→          address: form.address.trim(),
   598→          phone: form.phone.trim(),
   599→          email: form.email.trim(),
   600→          bankName: form.bankName.trim(),
   601→          bankAccount: form.bankAccount.trim(),
   602→          ifsc: form.ifsc.trim().toUpperCase(),
   603→          financialYear: form.financialYear,
   604→          invoicePrefix: form.invoicePrefix.trim() || undefined,
   605→          openingCash: Number(form.openingCash) || 0,
   606→          openingBank: Number(form.openingBank) || 0,
   607→        });
   608→        onOpenChange(false);
   609→        onSaved(res.firm.id);
   610→      } else if (firm) {
   611→        await apiPatch(`/api/v1/firms/${firm.id}`, {
   612→          firmName: form.firmName.trim(),
   613→          gstin: form.gstin.trim(),
   614→          state: stateName,
   615→          stateCode: form.state,
   616→          address: form.address.trim(),
   617→          phone: form.phone.trim(),
   618→          email: form.email.trim(),
   619→          bankName: form.bankName.trim(),
   620→          bankAccount: form.bankAccount.trim(),
   621→          ifsc: form.ifsc.trim().toUpperCase(),
   622→          financialYear: form.financialYear,
   623→          invoicePrefix: form.invoicePrefix.trim(),
   624→        });
   625→        onOpenChange(false);
   626→        onSaved();
   627→      }
   628→    } catch (e) {
   629→      const msg = e instanceof ApiError ? `${e.message}${e.code ? ` (${e.code})` : ""}` : "Request failed";
   630→      setFormError(msg);
   631→      toast({ variant: "destructive", title: mode === "create" ? "Could not create firm" : "Could not update firm", description: msg });
   632→    } finally {
   633→      setSaving(false);
   634→    }
   635→  }
   636→
   637→  return (
   638→    <Dialog open={open} onOpenChange={onOpenChange}>
   639→      <DialogContent className="max-w-2xl bg-dmk-bg-secondary border-dmk-border-medium max-h-[90vh] overflow-y-auto">
   640→        <DialogHeader>
   641→          <DialogTitle className="text-dmk-text-primary">
   642→            {mode === "create" ? "Create New Firm" : `Edit ${firm?.firmName ?? "Firm"}`}
   643→          </DialogTitle>
   644→          <DialogDescription className="text-dmk-text-muted text-[12px]">
   645→            {mode === "create"
   646→              ? "A new firm is a fresh isolated universe — chart of accounts is seeded automatically and the opening journal posts on save."
   647→              : "Profile fields feed invoices, A4 documents and GST place-of-supply logic. Firm code and opening balances are immutable."}
   648→          </DialogDescription>
   649→        </DialogHeader>
   650→
   651→        <div className="space-y-3">
   652→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   653→            <Field label="Firm Name *">
   654→              <Input value={form.firmName} onChange={(e) => set("firmName", e.target.value)} placeholder="DMK Mart" className={inputCls} />
   655→            </Field>
   656→            <Field label="Firm Code *" hint={mode === "edit" ? "Immutable after creation" : "Short unique code, e.g. DMK"}>
   657→              <Input
   658→                value={form.firmCode}
   659→                onChange={(e) => set("firmCode", e.target.value.toUpperCase())}
   660→                placeholder="DMK"
   661→                disabled={mode === "edit"}
   662→                className={cn(inputCls, "font-money uppercase")}
   663→              />
   664→            </Field>
   665→          </div>
   666→
   667→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   668→            <Field label="GSTIN" hint={gstinPrefix ? `First 2 digits “${gstinPrefix}” auto-map to state code` : "15 characters; first 2 digits = state code"}>
   669→              <Input
   670→                value={form.gstin}
   671→                onChange={(e) => set("gstin", e.target.value.toUpperCase())}
   672→                placeholder="27AAAAA0000A1Z5"
   673→                maxLength={15}
   674→                className={cn(inputCls, "font-money uppercase")}
   675→              />
   676→            </Field>
   677→            <Field label="State">
   678→              <Select
   679→                value={form.state}
   680→                onValueChange={(v) => {
   681→                  set("state", v);
   682→                  set("stateCode", v);
   683→                }}
   684→              >
   685→                <SelectTrigger className={cn(inputCls, "w-full")}>
   686→                  <SelectValue placeholder="Select state" />
   687→                </SelectTrigger>
   688→                <SelectContent className="bg-dmk-bg-tertiary border-dmk-border-medium max-h-64">
   689→                  {STATES.map((s) => (
   690→                    <SelectItem key={s.code} value={s.code} className="text-[12.5px]">
   691→                      {s.name}
   692→                    </SelectItem>
   693→                  ))}
   694→                </SelectContent>
   695→              </Select>
   696→            </Field>
   697→          </div>
   698→
   699→          <Field label="Address">
   700→            <Textarea value={form.address} onChange={(e) => set("address", e.target.value)} rows={2} placeholder="Shop no, street, city, PIN" className="bg-dmk-input-well border-dmk-border-subtle text-[13px] text-dmk-text-primary dmk-input resize-none" />
   701→          </Field>
   702→
   703→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   704→            <Field label="Phone">
   705→              <Input value={form.phone} onChange={(e) => set("phone", e.target.value)} placeholder="+91…" className={inputCls} />
   706→            </Field>
   707→            <Field label="Email">
   708→              <Input value={form.email} onChange={(e) => set("email", e.target.value)} placeholder="owner@firm.in" className={inputCls} />
   709→            </Field>
   710→          </div>
   711→
   712→          <div className="dmk-well p-3 space-y-3">
   713→            <p className="text-[11px] font-semibold uppercase tracking-wider text-dmk-text-muted">Bank Details (printed on A4 invoices)</p>
   714→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   715→              <Field label="Bank Name">
   716→                <Input value={form.bankName} onChange={(e) => set("bankName", e.target.value)} placeholder="HDFC Bank" className={inputCls} />
   717→              </Field>
   718→              <Field label="Account Number">
   719→                <Input value={form.bankAccount} onChange={(e) => set("bankAccount", e.target.value)} className={cn(inputCls, "font-money")} />
   720→              </Field>
   721→              <Field label="IFSC">
   722→                <Input value={form.ifsc} onChange={(e) => set("ifsc", e.target.value.toUpperCase())} className={cn(inputCls, "font-money uppercase")} />
   723→              </Field>
   724→            </div>
   725→          </div>
   726→
   727→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   728→            <Field label="Financial Year">
   729→              <Select value={form.financialYear} onValueChange={(v) => set("financialYear", v)}>
   730→                <SelectTrigger className={cn(inputCls, "w-full")}>
   731→                  <SelectValue />
   732→                </SelectTrigger>
   733→                <SelectContent className="bg-dmk-bg-tertiary border-dmk-border-medium">
   734→                  {FY_OPTIONS.map((fy) => (
   735→                    <SelectItem key={fy} value={fy} className="text-[12.5px]">
   736→                      FY {fy}
   737→                    </SelectItem>
   738→                  ))}
   739→                </SelectContent>
   740→              </Select>
   741→            </Field>
   742→            <Field label="Invoice Prefix" hint="Defaults to firm code when blank">
   743→              <Input
   744→                value={form.invoicePrefix}
   745→                onChange={(e) => set("invoicePrefix", e.target.value.toUpperCase())}
   746→                placeholder="DMK"
   747→                className={cn(inputCls, "font-money uppercase")}
   748→              />
   749→            </Field>
   750→          </div>
   751→
   752→          {mode === "create" && (
   753→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   754→              <Field label="Opening Cash (₹)" hint="Posts Dr Cash in the OPENING journal">
   755→                <Input
   756→                  type="number"
   757→                  min="0"
   758→                  step="0.01"
   759→                  value={form.openingCash}
   760→                  onChange={(e) => set("openingCash", e.target.value)}
   761→                  placeholder="0.00"
   762→                  className={cn(inputCls, "text-right font-money")}
   763→                />
   764→              </Field>
   765→              <Field label="Opening Bank (₹)" hint="Posts Dr Bank in the OPENING journal">
   766→                <Input
   767→                  type="number"
   768→                  min="0"
   769→                  step="0.01"
   770→                  value={form.openingBank}
   771→                  onChange={(e) => set("openingBank", e.target.value)}
   772→                  placeholder="0.00"
   773→                  className={cn(inputCls, "text-right font-money")}
   774→                />
   775→              </Field>
   776→            </div>
   777→          )}
   778→
   779→          {formError && <ErrorText>{formError}</ErrorText>}
   780→        </div>
   781→
   782→        <DialogFooter>
   783→          <Button variant="outline" onClick={() => onOpenChange(false)} className="h-9 border-dmk-border-subtle bg-transparent hover:bg-dmk-hover">
   784→            Cancel
   785→          </Button>
   786→          <Button
   787→            onClick={() => void save()}
   788→            disabled={saving || (mode === "create" ? !createValid : !editValid)}
   789→            className="h-9 gap-2 bg-dmk-orange text-white hover:bg-dmk-orange/85 disabled:opacity-40"
   790→          >
   791→            {saving && <Loader2 className="h-3.5 w-3.5 animate-spin" />}
   792→            {mode === "create" ? "Create Firm" : "Save Changes"}
   793→          </Button>
   794→        </DialogFooter>
   795→      </DialogContent>
   796→    </Dialog>
   797→  );
   798→}
