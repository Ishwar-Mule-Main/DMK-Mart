   150→   150→function gstPreview(taxable: number, gstRate: number, intra: boolean) {
   151→   151→  if (intra) {
   152→   152→    return { cgst: round2(taxable * (gstRate / 200)), sgst: round2(taxable * (gstRate / 200)), igst: 0 };
   153→   153→  }
   154→   154→  return { cgst: 0, sgst: 0, igst: round2(taxable * (gstRate / 100)) };
   155→   155→}
