   108→    if (!activeFirmId) return;

   111→      try {
   112→        const res = await apiGet<Customer[]>("/api/v1/customers", { firmId: activeFirmId, type: "B2B", search: query.trim() });

   221→// ═══════════════════════════════════════════════════════════════
   222→// B2B create / edit dialog
   223→// ═══════════════════════════════════════════════════════════════
   224→function B2BFormDialog({
   225→  open,
   226→  onOpenChange,
   227→  editing,
   228→  firmStateCode,
   229→  onSaved,
   230→  onRefresh,
   231→}: {
   232→  open: boolean;
   233→  onOpenChange: (o: boolean) => void;
   234→  editing: Customer | null;
   235→  firmStateCode: string;
   236→  onSaved: () => void;
   237→  onRefresh: () => void;
   238→}) {
   239→  const { toast } = useToast();
   240→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   241→  const [saving, setSaving] = React.useState(false);
   242→  const [form, setForm] = React.useState({
   243→    city: "",
   244→    firmName: "",
   245→    gstin: "",
   246→    phone: "",
   247→    email: "",
   248→    address: "",
   249→    stateCode: firmStateCode,
   250→    assignedTier: "tier3SemiWholesale",
   251→    creditLimit: "0",
   252→    creditDays: "30",
   253→    openingBalance: "0",
   254→  });
   255→
   256→  React.useEffect(() => {
   257→    if (!open) return;
   258→    if (editing) {
   259→      setForm({
   260→        city: editing.city ?? "",
   261→        firmName: editing.firmName ?? "",
   262→        gstin: editing.gstin ?? "",
   263→        phone: editing.phone ?? "",
   264→        email: editing.email ?? "",
   265→        address: editing.address ?? "",
   266→        stateCode: editing.stateCode || firmStateCode,
   267→        assignedTier: editing.assignedTier || "tier3SemiWholesale",
   268→        creditLimit: String(editing.creditLimit ?? 0),
   269→        creditDays: String(editing.creditDays ?? 30),
   270→        openingBalance: "0",
   271→      });
   272→    } else {
   273→      setForm({
   274→        city: "",
   275→        firmName: "",
   276→        gstin: "",
   277→        phone: "",
   278→        email: "",
   279→        address: "",
   280→        stateCode: firmStateCode,
   281→        assignedTier: "tier3SemiWholesale",
   282→        creditLimit: "0",
   283→        creditDays: "30",
   284→        openingBalance: "0",
   285→      });
   286→    }
   287→  }, [open, editing, firmStateCode]);
   288→
   289→  const set = (k: keyof typeof form) => (e: React.ChangeEvent<HTMLInputElement>) => setForm((f) => ({ ...f, [k]: e.target.value }));
   290→
   291→  async function submit() {
   292→    if (!activeFirmId) return;
   293→    if (!form.city.trim() || !form.firmName.trim()) {
   294→      toast({ variant: "destructive", title: "Missing fields", description: "City and firm name are required — party name becomes “City FirmName”." });
   295→      return;
   296→    }
   297→    setSaving(true);
   298→    try {
   299→      if (editing) {
   300→        await apiPatch(`/api/v1/customers/${editing.id}`, {
   301→          city: form.city.trim(),
   302→          firmName: form.firmName.trim(),
   303→          gstin: form.gstin.trim(),
   304→          phone: form.phone.trim(),
   305→          email: form.email.trim(),
   306→          address: form.address.trim(),
   307→          stateCode: form.stateCode,
   308→          assignedTier: form.assignedTier,
   309→          creditLimit: Number(form.creditLimit) || 0,
   310→          creditDays: Number(form.creditDays) || 0,
   311→        });
   312→        toast({ title: "Customer updated", description: `${form.city.trim()} ${form.firmName.trim()}` });
   313→      } else {
   314→        await apiPost<Customer>("/api/v1/customers", {
   315→          firmId: activeFirmId,
   316→          customerType: "B2B",
   317→          city: form.city.trim(),
   318→          firmName: form.firmName.trim(),
   319→          gstin: form.gstin.trim(),
   320→          phone: form.phone.trim(),
   321→          email: form.email.trim(),
   322→          address: form.address.trim(),
   323→          stateCode: form.stateCode,
   324→          assignedTier: form.assignedTier,
   325→          creditLimit: Number(form.creditLimit) || 0,
   326→          creditDays: Number(form.creditDays) || 0,
   327→          openingBalance: Number(form.openingBalance) || 0,
   328→        });
   329→        toast({ title: "Customer created", description: `${form.city.trim()} ${form.firmName.trim()} added to the trade directory.` });
   330→      }
   331→      onRefresh();
   332→      onSaved();
   333→      onOpenChange(false);
   334→    } catch (e) {
   335→      const msg = e instanceof ApiError ? e.message : "Could not save customer.";
   336→      toast({ variant: "destructive", title: "Save failed", description: msg });
   337→    } finally {
   338→      setSaving(false);
   339→    }
   340→  }
   341→
   342→  return (
   343→    <Dialog open={open} onOpenChange={onOpenChange}>
   344→      <DialogContent className="sm:max-w-xl dmk-elevated border-dmk-border-medium max-h-[90vh] overflow-y-auto">
   345→        <DialogHeader>
   346→          <DialogTitle className="text-dmk-text-primary">{editing ? "Edit B2B customer" : "New B2B customer"}</DialogTitle>
   347→          <DialogDescription className="text-dmk-text-muted">Location-first naming (R8) — party name is composed as “City FirmName”.</DialogDescription>
   348→        </DialogHeader>
   349→        <div className="grid grid-cols-2 gap-3">
   350→          <Field label="City *">
   351→            <Input value={form.city} onChange={set("city")} placeholder="e.g. Latur" className={inputCls} />
   352→          </Field>
   353→          <Field label="Firm name *">
   354→            <Input value={form.firmName} onChange={set("firmName")} placeholder="e.g. Ishwar Mule" className={inputCls} />
   355→          </Field>
   356→          <Field label="GSTIN">
   357→            <Input value={form.gstin} onChange={set("gstin")} placeholder="27…" className={cn(inputCls, "font-money")} />
   358→          </Field>
   359→          <Field label="Phone">
   360→            <Input value={form.phone} onChange={set("phone")} className={cn(inputCls, "font-money")} />
   361→          </Field>
   362→          <Field label="State code">
   363→            <Select value={form.stateCode} onValueChange={(v) => setForm((f) => ({ ...f, stateCode: v }))}>
   364→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   365→              <SelectContent className="max-h-64">
   366→                {STATES.map((s) => (
   367→                  <SelectItem key={s.code} value={s.code}>{s.name}</SelectItem>
   368→                ))}
   369→              </SelectContent>
   370→            </Select>
   371→          </Field>
   372→          <Field label="Assigned tier">
   373→            <Select value={form.assignedTier} onValueChange={(v) => setForm((f) => ({ ...f, assignedTier: v }))}>
   374→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   375→              <SelectContent>
   376→                {TIERS.slice(0, 4).map((t) => (
   377→                  <SelectItem key={t.key} value={t.key}>{t.label}</SelectItem>
   378→                ))}
   379→              </SelectContent>
   380→            </Select>
   381→          </Field>
   382→          <Field label="Credit limit ₹">
   383→            <Input type="number" min={0} value={form.creditLimit} onChange={set("creditLimit")} className={cn(inputCls, "font-money")} />
   384→          </Field>
   385→          <Field label="Credit days">
   386→            <Input type="number" min={0} value={form.creditDays} onChange={set("creditDays")} className={cn(inputCls, "font-money")} />
   387→          </Field>
   388→          {!editing && (
   389→            <>
   390→              <Field label="Opening balance ₹ (Dr +)" hint="Positive = receivable (Dr). Posts an OPENING ledger row.">
   391→                <Input type="number" step="0.01" value={form.openingBalance} onChange={set("openingBalance")} className={cn(inputCls, "font-money")} />
   392→              </Field>
   393→              <Field label="Email">
   394→                <Input value={form.email} onChange={set("email")} className={inputCls} placeholder="optional" />
   395→              </Field>
   396→            </>
   397→          )}
   398→          {editing && (
   399→            <Field label="Email">
   400→              <Input value={form.email} onChange={set("email")} className={inputCls} placeholder="optional" />
   401→            </Field>
   402→          )}
   403→          <div className="col-span-2">
   404→            <Field label="Address">
   405→              <Input value={form.address} onChange={set("address")} className={inputCls} placeholder="optional" />
   406→            </Field>
   407→          </div>
   408→        </div>
   409→        <DialogFooter>
   410→          <Button variant="outline" onClick={() => onOpenChange(false)} className="border-dmk-border-medium text-dmk-text-secondary hover:bg-dmk-hover">Cancel</Button>
   411→          <Button onClick={submit} disabled={saving} className="bg-dmk-orange text-white hover:bg-dmk-orange/90">
   412→            {saving && <Loader2 className="h-4 w-4 animate-spin" />} {editing ? "Save changes" : "Create customer"}
   413→          </Button>
   414→        </DialogFooter>
   415→      </DialogContent>
   416→    </Dialog>
   417→  );
   418→}

   421→// Ledger dialog — opening + last 100 rows + closing
   422→// ═══════════════════════════════════════════════════════════════
   423→function LedgerDialog({ customer, onClose }: { customer: Customer | null; onClose: () => void }) {
   424→  const [ledger, setLedger] = React.useState<LedgerRow[] | null>(null);
   425→
   426→  React.useEffect(() => {
   427→    if (!customer) return;
   428→    let alive = true;
   429→    setLedger(null);
   430→    apiGet<{ customer: Customer; ledger: LedgerRow[] }>(`/api/v1/customers/${customer.id}`)
   431→      .then((res) => alive && setLedger(res.ledger))
   432→      .catch(() => alive && setLedger([]));
   433→    return () => {
   434→      alive = false;
   435→    };
   436→  }, [customer]);
   437→
   438→  const closing = Number(customer?.closingBalance ?? 0);
   439→
   440→  return (
   441→    <Dialog open={!!customer} onOpenChange={(o) => !o && onClose()}>
   442→      <DialogContent className="sm:max-w-2xl dmk-elevated border-dmk-border-medium">
   443→        <DialogHeader>
   444→          <DialogTitle className="text-dmk-text-primary">Party ledger — {customer?.partyName}</DialogTitle>
   445→          <DialogDescription className="text-dmk-text-muted">
   446→            GSTIN <span className="font-money">{customer?.gstin || "—"}</span> · Credit {formatINR(Number(customer?.creditLimit ?? 0))} / {customer?.creditDays ?? 0} days
   447→          </DialogDescription>
   448→        </DialogHeader>
   449→
   450→        <div className="flex items-center justify-between dmk-well px-3 py-2.5 text-[12.5px]">
   451→          <span className="text-dmk-text-muted">Opening <span className="font-money text-dmk-text-secondary">{formatINR(Number(customer?.openingBalance ?? 0))}</span></span>
   452→          <span className="text-dmk-text-muted">
   453→            Closing{" "}
   454→            <span className={cn("font-money font-semibold", closing > 0.005 ? "text-dmk-orange" : "text-dmk-success")}>
   455→              {closing > 0.005 ? `Dr ${formatINR(closing)}` : "Clear"}
   456→            </span>
   457→          </span>
   458→        </div>
   459→
   460→        <div className="dmk-card overflow-hidden max-h-80 overflow-y-auto">
   461→          {ledger === null ? (
   462→            <LoadingRows rows={5} />
   463→          ) : ledger.length === 0 ? (
   464→            <EmptyState icon={BookOpen} title="No ledger rows yet" hint="Invoices, credit notes and receipts will appear here." />
   465→          ) : (
   466→            <table className="dmk-table">
   467→              <thead>
   468→                <tr>
   469→                  <th>Date</th>
   470→                  <th>Voucher</th>
   471→                  <th>Voucher #</th>
   472→                  <th>Particulars</th>
   473→                  <th className="text-right">Dr</th>
   474→                  <th className="text-right">Cr</th>
   475→                  <th className="text-right">Balance</th>
   476→                </tr>
   477→              </thead>
   478→              <tbody>
   479→                {ledger.map((r) => (
   480→                  <tr key={r.id}>
   481→                    <td className="text-[11.5px] text-dmk-text-secondary whitespace-nowrap">{formatDate(r.entryDate)}</td>
   482→                    <td><StatusBadge status={r.voucherType} /></td>

   589→function NewBuyerDialog({ open, onOpenChange, onCreated }: { open: boolean; onOpenChange: (o: boolean) => void; onCreated: () => void }) {
   590→  const { toast } = useToast();
   591→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   592→  const firm = useActiveFirm();
   593→  const [name, setName] = React.useState("");
   594→  const [phone, setPhone] = React.useState("");
   595→  const [saving, setSaving] = React.useState(false);
   596→
   597→  async function submit() {
   598→    if (!activeFirmId) return;
   599→    if (!name.trim()) {
   600→      toast({ variant: "destructive", title: "Name required" });
   601→      return;
   602→    }
   603→    setSaving(true);
   604→    try {
   605→      await apiPost<Customer>("/api/v1/customers", {
   606→        firmId: activeFirmId,
   607→        customerType: "B2C_COUNTER",
   608→        firmName: name.trim(),
   609→        phone: phone.trim(),
   610→        city: "",
   611→        stateCode: firm?.stateCode ?? "",
   612→        assignedTier: "tier4Retailer",
   613→        creditLimit: 0,
   614→        creditDays: 0,
   615→      });
   616→      toast({ title: "Buyer added", description: name.trim() });
   617→      setName("");
   618→      setPhone("");
   619→      onCreated();
   620→      onOpenChange(false);
   621→    } catch (e) {
   622→      toast({ variant: "destructive", title: "Create failed", description: e instanceof ApiError ? e.message : "Unknown error" });
   623→    } finally {
   624→      setSaving(false);
   625→    }
   626→  }
   627→
   628→  return (
   629→    <Dialog open={open} onOpenChange={onOpenChange}>
   630→      <DialogContent className="sm:max-w-sm dmk-elevated border-dmk-border-medium">
   631→        <DialogHeader>
   632→          <DialogTitle className="text-dmk-text-primary">New counter buyer</DialogTitle>
   633→          <DialogDescription className="text-dmk-text-muted">No credit — instant payment only (R14).</DialogDescription>
   634→        </DialogHeader>
   635→        <div className="space-y-3">
   636→          <Field label="Name *">
   637→            <Input value={name} onChange={(e) => setName(e.target.value)} className={inputCls} autoFocus />
   638→          </Field>
   639→          <Field label="Phone">
   640→            <Input value={phone} onChange={(e) => setPhone(e.target.value)} className={cn(inputCls, "font-money")} />
   641→          </Field>
   642→        </div>
   643→        <DialogFooter>
   644→          <Button variant="outline" onClick={() => onOpenChange(false)} className="border-dmk-border-medium text-dmk-text-secondary hover:bg-dmk-hover">Cancel</Button>
   645→          <Button onClick={submit} disabled={saving} className="bg-dmk-orange text-white hover:bg-dmk-orange/90">
   646→            {saving && <Loader2 className="h-4 w-4 animate-spin" />} Add buyer
   647→          </Button>
   648→        </DialogFooter>
   649→      </DialogContent>
   650→    </Dialog>
   651→  );
   652→}
