    95→    95→function previewGst(taxable: number, rate: number, seller: string, buyer: string) {
    96→    96→  const intra = seller && buyer && seller === buyer;
    97→    97→  if (intra) return { cgst: round2((taxable * rate) / 200), sgst: round2((taxable * rate) / 200), igst: 0 };
    98→    98→  return { cgst: 0, sgst: 0, igst: round2((taxable * rate) / 100) };
    99→    99→}
