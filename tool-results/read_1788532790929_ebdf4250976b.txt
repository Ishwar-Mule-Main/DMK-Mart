    91→function round2(n: number): number {
    92→  return Math.round((n + Number.EPSILON) * 100) / 100;

    95→function num(s: string): number {
    96→  const n = Number(s);
    97→  return Number.isFinite(n) ? n : 0;

   100→function normalizeProducts(res: unknown): Product[] {
   101→  if (Array.isArray(res)) return res as Product[];
   102→  const obj = res as { products?: Product[] } | null;
   103→  return obj?.products ?? [];
   104→}
   105→
   106→function gstPreview(taxable: number, gstRate: number, intra: boolean) {
   107→  if (intra) {
   108→    return { cgst: round2(taxable * (gstRate / 200)), sgst: round2(taxable * (gstRate / 200)), igst: 0 };
   109→  }
   110→  return { cgst: 0, sgst: 0, igst: round2(taxable * (gstRate / 100)) };
   111→}

   330→const SETTLEMENT_MODES: Array<{ value: SettlementMode; short: string; sub: string }> = [
   331→  { value: "CREDIT", short: "Adjust in credit", sub: "Amount adjusts the vendor's payable (opening balance) — credited against what we owe them" },
   332→  { value: "UPI_NEFT", short: "UPI / NEFT", sub: "Vendor pays the amount back instantly via bank transfer" },

   336→interface PoLite {

   338→  poNumber: string;
   339→  poDate: string;

   344→
   345→function NewDebitNoteDialog({
   346→  open,
   347→  onOpenChange,
   348→  onSaved,
   349→}: {
   350→  open: boolean;
   351→  onOpenChange: (o: boolean) => void;
   352→  onSaved: () => void;
   353→}) {
   354→  const { toast } = useToast();
   355→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   356→  const firm = useActiveFirm();
   357→  const firmStateCode = firm?.stateCode ?? "27";
   358→
   359→  const [saving, setSaving] = React.useState(false);
   360→  const [vendors, setVendors] = React.useState<Vendor[]>([]);
   361→  const [products, setProducts] = React.useState<Product[]>([]);
   362→  const [vendorId, setVendorId] = React.useState("NONE");
   363→  const [poId, setPoId] = React.useState("NONE");
   364→  const [poOptions, setPoOptions] = React.useState<PoLite[]>([]);
   365→  const [poSort, setPoSort] = React.useState<"recent" | "oldest">("recent");
   366→  const [settlementMode, setSettlementMode] = React.useState<SettlementMode>("CREDIT");
   367→  const [returnDate, setReturnDate] = React.useState(toISODate(new Date()));
   368→  const [lines, setLines] = React.useState<DnLine[]>([]);
   369→
   370→  React.useEffect(() => {
   371→    if (!open || !activeFirmId) return;
   372→    let alive = true;
   373→    apiGet<Vendor[]>("/api/v1/vendors", { firmId: activeFirmId })
   374→      .then((res) => alive && setVendors(res))
   375→      .catch(() => alive && setVendors([]));
   376→    apiGet<Product[]>("/api/v1/products", { firmId: activeFirmId })
   377→      .then((res) => alive && setProducts(normalizeProducts(res)))
   378→      .catch(() => alive && setProducts([]));
   379→    return () => {
   380→      alive = false;
   381→    };
   382→  }, [open, activeFirmId]);
   383→
   384→  // Load the vendor's CONFIRMED POs for the reference select
   385→  React.useEffect(() => {
   386→    if (!open || !activeFirmId || vendorId === "NONE") {
   387→      setPoOptions([]);
   388→      setPoId("NONE");
   389→      return;
   390→    }
   391→    let alive = true;
   392→    apiGet<PoLite[]>("/api/v1/purchase-orders", { firmId: activeFirmId, status: "CONFIRMED" })
   393→      .then((res) => {
   394→        if (!alive) return;
   395→        const mine = res.filter((p) => p.vendorId === vendorId);
   396→        setPoOptions(mine);
   397→        setPoId("NONE");
   398→      })
   399→      .catch(() => alive && setPoOptions([]));
   400→    return () => {
   401→      alive = false;
   402→    };
   403→  }, [open, activeFirmId, vendorId]);
   404→
   405→  // Recent POs first, flip with the sort toggle
   406→  const sortedPoOptions = React.useMemo(() => {
   407→    const arr = [...poOptions];
   408→    arr.sort((a, b) => {
   409→      const da = new Date(a.poDate).getTime() || 0;
   410→      const dbb = new Date(b.poDate).getTime() || 0;
   411→      return poSort === "recent" ? dbb - da : da - dbb;
   412→    });
   413→    return arr;
   414→  }, [poOptions, poSort]);
   415→
   416→  React.useEffect(() => {
   417→    if (!open) return;
   418→    setVendorId("NONE");
   419→    setPoId("NONE");
   420→    setPoOptions([]);
   421→    setPoSort("recent");
   422→    setSettlementMode("CREDIT");
   423→    setReturnDate(toISODate(new Date()));
   424→    setLines([]);
   425→  }, [open]);
   426→
   427→  // Selecting a PO auto-loads its lines at the PO prices — every row
   428→  // starts with damaged qty 0 (not returned) until typed in.
   429→  React.useEffect(() => {
   430→    if (!open || poId === "NONE") return;
   431→    const po = poOptions.find((p) => p.id === poId);
   432→    if (!po) return;
   433→    setLines(
   434→      po.items.map((it) => ({
   435→        productId: it.productId,
   436→        qty: "0",
   437→        cost: String(it.unitCost ?? 0),
   438→        reason: "Transit Damage",
   439→        poQty: Number(it.quantity) || 0,
   440→        fromPo: true,
   441→      })),
   442→    );
   443→  }, [poId, open]);
   444→
   445→  const vendor = vendors.find((v) => v.id === vendorId);
   446→  // No vendor → seller state falls back to the firm's own state (intra, per server)
   447→  const intra = !vendor || vendor.stateCode === firmStateCode;
   448→  const refPo = poOptions.find((p) => p.id === poId);
   449→
   450→  const productMap = new Map(products.map((p) => [p.id, p]));
   451→
   452→  const computed = lines.map((l) => {
   453→    const p = productMap.get(l.productId);
   454→    const taxable = round2(num(l.qty) * num(l.cost));
   455→    const gst = gstPreview(taxable, p?.gstRate ?? 0, intra);
   456→    const overStock = !!p && num(l.qty) > p.damagedStock + 0.001;
   457→    return { line: l, product: p, taxable, gst, total: round2(taxable + gst.cgst + gst.sgst + gst.igst), overStock };
   458→  });
   459→
   460→  const totals = computed.reduce(
   461→    (acc, c) => ({
   462→      taxable: round2(acc.taxable + c.taxable),
   463→      cgst: round2(acc.cgst + c.gst.cgst),
   464→      sgst: round2(acc.sgst + c.gst.sgst),
   465→      igst: round2(acc.igst + c.gst.igst),
   466→    }),
   467→    { taxable: 0, cgst: 0, sgst: 0, igst: 0 }
   468→  );
   469→  const grand = round2(totals.taxable + totals.cgst + totals.sgst + totals.igst);
   470→
   471→  const stockErrors = computed.filter((c) => c.overStock);
   472→  const returnRows = computed.filter((c) => num(c.line.qty) > 0);
   473→  const overPoRows = returnRows.filter((c) => c.line.poQty != null && num(c.line.qty) > (c.line.poQty ?? 0));
   474→  const linesValid = returnRows.length > 0 && returnRows.every((c) => num(c.line.qty) > 0 && num(c.line.cost) >= 0);
   475→  const canSave = linesValid && stockErrors.length === 0 && overPoRows.length === 0;
   476→
   477→  function addProduct(p: Product) {
   478→    setLines((ls) => {
   479→      const idx = ls.findIndex((l) => l.productId === p.id);
   480→      if (idx >= 0) {
   481→        const next = [...ls];
   482→        next[idx] = { ...next[idx], qty: String(num(next[idx].qty) + 1) };
   483→        return next;
   484→      }
   485→      return [...ls, { productId: p.id, qty: "1", cost: String(p.purchaseCost), reason: "Transit Damage" }];
   486→    });
   487→  }
   488→
   489→  function updateLine(i: number, patch: Partial<DnLine>) {
   490→    setLines((ls) => ls.map((l, idx) => (idx === i ? { ...l, ...patch } : l)));
   491→  }
   492→
   493→  function removeLine(i: number) {
   494→    setLines((ls) => ls.filter((_, idx) => idx !== i));
   495→  }
   496→
   497→  async function submit() {
   498→    if (!activeFirmId || !canSave) return;
   499→    setSaving(true);
   500→    try {
   501→      const res = await apiPost<{ purchaseReturn: PrRow; journal: unknown; settlementMode: SettlementMode }>("/api/v1/purchase-returns", {
   502→        firmId: activeFirmId,
   503→        vendorId: vendorId === "NONE" ? undefined : vendorId,
   504→        poId: poId === "NONE" ? undefined : poId,
   505→        returnDate,
   506→        settlementMode,
