    97→    97→function defectTone(d: string): "danger" | "warning" | "info" | "neutral" {
    98→    98→  switch (d) {
    99→    99→    case "Broken": return "danger";
   100→   100→    case "Damaged": return "warning";
   101→   101→    case "Defective": return "info";
   102→   102→    default: return "neutral";
   103→   103→  }

   106→   106→function previewGst(taxable: number, rate: number, seller: string, buyer: string) {
   107→   107→  const intra = seller && buyer && seller === buyer;
   108→   108→  if (intra) return { cgst: round2((taxable * rate) / 200), sgst: round2((taxable * rate) / 200), igst: 0 };
   109→   109→  return { cgst: 0, sgst: 0, igst: round2((taxable * rate) / 100) };
   110→   110→}

   435→   435→function NewReturnDialog({
   436→   436→  open,
   437→   437→  onOpenChange,
   438→   438→  onCreated,
   439→   439→  firmStateCode,
   440→   440→}: {
   441→   441→  open: boolean;
   442→   442→  onOpenChange: (o: boolean) => void;
   443→   443→  onCreated: () => void;
   444→   444→  firmStateCode: string;

   446→   446→  const { toast } = useToast();
   447→   447→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   448→   448→
   449→   449→  const [customers, setCustomers] = React.useState<Customer[]>([]);
   450→   450→  const [products, setProducts] = React.useState<Product[]>([]);
   451→   451→  const [customerId, setCustomerId] = React.useState("");
   452→   452→  const [invoiceId, setInvoiceId] = React.useState("");
   453→   453→  const [custInvoices, setCustInvoices] = React.useState<Invoice[]>([]);
   454→   454→  const [returnDate, setReturnDate] = React.useState(toISODate(new Date()));

   464→   464→          apiGet<Customer[]>("/api/v1/customers", { firmId: activeFirmId }),
   465→   465→          apiGet<Product[] | { products: Product[] }>("/api/v1/products", { firmId: activeFirmId, activeOnly: "true" }),

   472→   472→        if (alive && e instanceof ApiError) toast({ variant: "destructive", title: "Could not load data", description: e.message });
   473→   473→      }
   474→   474→    })();

   476→   476→      alive = false;
   477→   477→    };

   489→   489→      .then((res) => alive && setCustInvoices(res))
   490→   490→      .catch(() => alive && setCustInvoices([]));

   492→   492→      alive = false;
   493→   493→    };

   496→   496→  function addItem() {
   497→   497→    if (products.length === 0) return;
   498→   498→    const p = products[0];
   499→   499→    setItems((prev) => [...prev, { productId: p.id, qty: 1, defectType: "Damaged", unitPrice: Number(p.tier4Retailer) || 0 }]);
   500→   500→  }
   501→   501→
   502→   502→  function updateItem(idx: number, patch: Partial<DraftItem>) {
   503→   503→    setItems((prev) => prev.map((it, i) => (i === idx ? { ...it, ...patch } : it)));
   504→   504→  }
   505→   505→
   506→   506→  const draftTotals = React.useMemo(() => {
   507→   507→    let taxable = 0;
   508→   508→    let tax = 0;
   509→   509→    for (const it of items) {
   510→   510→      const p = products.find((x) => x.id === it.productId);
   511→   511→      if (!p) continue;
   512→   512→      const t = round2(Number(it.unitPrice) * it.qty);
   513→   513→      taxable += t;
   514→   514→      const g = previewGst(t, p.gstRate, firmStateCode, customers.find((c) => c.id === customerId)?.stateCode ?? firmStateCode);
   515→   515→      tax += g.cgst + g.sgst + g.igst;
   516→   516→    }
   517→   517→    return { taxable: round2(taxable), tax: round2(tax), grand: round2(round2(taxable) + round2(tax)) };
   518→   518→  }, [items, products, customerId, customers, firmStateCode]);
   519→   519→
   520→   520→  const selectedInvoice = custInvoices.find((i) => i.id === invoiceId);

   537→   537→        customerId: customerId || undefined,
   538→   538→        invoiceId: invoiceId || undefined,

   543→   543→      });

   546→   546→      setCustomerId("");
   547→   547→      setInvoiceId("");
   548→   548→      onCreated();
   549→   549→      onOpenChange(false);

   558→   558→  return (
