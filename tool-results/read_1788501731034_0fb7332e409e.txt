    97→function defectTone(d: string): "danger" | "warning" | "info" | "neutral" {
    98→  switch (d) {
    99→    case "Broken": return "danger";
   100→    case "Damaged": return "warning";
   101→    case "Defective": return "info";
   102→    default: return "neutral";
   103→  }

   106→function previewGst(taxable: number, rate: number, seller: string, buyer: string) {
   107→  const intra = seller && buyer && seller === buyer;
   108→  if (intra) return { cgst: round2((taxable * rate) / 200), sgst: round2((taxable * rate) / 200), igst: 0 };
   109→  return { cgst: 0, sgst: 0, igst: round2((taxable * rate) / 100) };
   110→}

   435→function NewReturnDialog({
   436→  open,
   437→  onOpenChange,

   439→  firmStateCode,
   440→}: {
   441→  open: boolean;
   442→  onOpenChange: (o: boolean) => void;
   443→  onCreated: () => void;
   444→  firmStateCode: string;
   445→}) {
   446→  const { toast } = useToast();
   447→  const activeFirmId = useErpStore((s) => s.activeFirmId);
   448→
   449→  const [customers, setCustomers] = React.useState<Customer[]>([]);
   450→  const [products, setProducts] = React.useState<Product[]>([]);
   451→  const [customerId, setCustomerId] = React.useState("");
   452→  const [invoiceId, setInvoiceId] = React.useState("");
   453→  const [custInvoices, setCustInvoices] = React.useState<Invoice[]>([]);
   454→  const [returnDate, setReturnDate] = React.useState(toISODate(new Date()));
   455→  const [items, setItems] = React.useState<DraftItem[]>([]);
   456→  const [saving, setSaving] = React.useState(false);
   457→
   458→  React.useEffect(() => {
   459→    if (!open || !activeFirmId) return;
   460→    let alive = true;
   461→    (async () => {
   462→      try {
   463→        const [c, p] = await Promise.all([
   464→          apiGet<Customer[]>("/api/v1/customers", { firmId: activeFirmId }),
   465→          apiGet<Product[] | { products: Product[] }>("/api/v1/products", { firmId: activeFirmId, activeOnly: "true" }),
   466→        ]);
   467→        if (alive) {
   468→          setCustomers(c);
   469→          setProducts(Array.isArray(p) ? p : (p.products ?? []));
   470→        }
   471→      } catch (e) {
   472→        if (alive && e instanceof ApiError) toast({ variant: "destructive", title: "Could not load data", description: e.message });
   473→      }
   474→    })();
   475→    return () => {
   476→      alive = false;
   477→    };
   478→  }, [open, activeFirmId]);
   479→
   480→  // Load the selected customer's invoices for the reference picker
   481→  React.useEffect(() => {
   482→    if (!open || !activeFirmId || !customerId) {
   483→      setCustInvoices([]);
   484→      setInvoiceId("");
   485→      return;
   486→    }
   487→    let alive = true;
   488→    apiGet<Invoice[]>("/api/v1/invoices", { firmId: activeFirmId, customerId })
   489→      .then((res) => alive && setCustInvoices(res))
   490→      .catch(() => alive && setCustInvoices([]));
   491→    return () => {
   492→      alive = false;
   493→    };
   494→  }, [customerId, activeFirmId, open]);
   495→
   496→  function addItem() {
   497→    if (products.length === 0) return;
   498→    const p = products[0];
   499→    setItems((prev) => [...prev, { productId: p.id, qty: 1, defectType: "Damaged", unitPrice: Number(p.tier4Retailer) || 0 }]);
   500→  }
   501→
   502→  function updateItem(idx: number, patch: Partial<DraftItem>) {
   503→    setItems((prev) => prev.map((it, i) => (i === idx ? { ...it, ...patch } : it)));
   504→  }
   505→
   506→  const draftTotals = React.useMemo(() => {
   507→    let taxable = 0;
   508→    let tax = 0;
   509→    for (const it of items) {
   510→      const p = products.find((x) => x.id === it.productId);
   511→      if (!p) continue;
   512→      const t = round2(Number(it.unitPrice) * it.qty);
   513→      taxable += t;
   514→      const g = previewGst(t, p.gstRate, firmStateCode, customers.find((c) => c.id === customerId)?.stateCode ?? firmStateCode);
   515→      tax += g.cgst + g.sgst + g.igst;
   516→    }
   517→    return { taxable: round2(taxable), tax: round2(tax), grand: round2(round2(taxable) + round2(tax)) };
   518→  }, [items, products, customerId, customers, firmStateCode]);
   519→
   520→  const selectedInvoice = custInvoices.find((i) => i.id === invoiceId);
   521→
   522→  async function submit() {
   523→    if (!activeFirmId) return;
   524→    if (items.length === 0) {
   525→      toast({ variant: "destructive", title: "No items", description: "Add at least one returned product." });
   526→      return;
   527→    }
   528→    const valid = items.every((it) => it.productId && it.qty > 0 && Number(it.unitPrice) >= 0);
   529→    if (!valid) {
   530→      toast({ variant: "destructive", title: "Invalid rows", description: "Every line needs a product, positive qty and a rate." });
   531→      return;
   532→    }
   533→    setSaving(true);
   534→    try {
   535→      await apiPost("/api/v1/sales-returns", {
   536→        firmId: activeFirmId,
   537→        customerId: customerId || undefined,
   538→        invoiceId: invoiceId || undefined,
   539→        invoiceRef: selectedInvoice?.invoiceNumber ?? "",
   540→        returnDate,
   541→        notes: "",
   542→        items: items.map((it) => ({ productId: it.productId, damagedQty: it.qty, unitPrice: Number(it.unitPrice), defectType: it.defectType })),
   543→      });
   544→      toast({ title: "Return recorded", description: "Credit note posted — qty moved to Damaged Stock." });
   545→      setItems([]);
   546→      setCustomerId("");
   547→      setInvoiceId("");
   548→      onCreated();
   549→      onOpenChange(false);
   550→    } catch (e) {
   551→      const msg = e instanceof ApiError ? e.message : "Could not record return.";
   552→      toast({ variant: "destructive", title: "Return failed", description: msg });
   553→    } finally {
   554→      setSaving(false);
   555→    }
   556→  }
   557→
   558→  return (
   559→    <Dialog open={open} onOpenChange={onOpenChange}>
   560→      <DialogContent className="dmk-elevated border-dmk-border-medium">
   561→        <DialogHeader>
   562→          <DialogTitle className="text-dmk-text-primary">New sales return</DialogTitle>
   563→          <DialogDescription className="text-dmk-text-muted">
   564→            Returned qty is quarantined to Damaged Stock — never sellable again. Credit note reduces the customer receivable.
   565→          </DialogDescription>
   566→        </DialogHeader>
   567→
   568→        <div className="grid grid-cols-2 gap-3">
   569→          <Field label="Customer">
   570→            <Select
   571→              value={customerId}
   572→              onValueChange={(v) => setCustomerId(v)}
   573→            >
   574→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue placeholder="Select customer" /></SelectTrigger>
   575→              <SelectContent className="max-h-64">
   576→                {customers.map((c) => (
   577→                  <SelectItem key={c.id} value={c.id}>
   578→                    {c.partyName} {c.customerType === "B2C_COUNTER" ? "(counter)" : ""}
   579→                  </SelectItem>
   580→                ))}
   581→              </SelectContent>
   582→            </Select>
   583→          </Field>
   584→          <Field label="Invoice reference (optional)">
   585→            <Select value={invoiceId} onValueChange={setInvoiceId} disabled={!customerId}>
   586→              <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue placeholder={customerId ? "Select invoice" : "Pick a customer first"} /></SelectTrigger>
   587→              <SelectContent className="max-h-64">
   588→                {custInvoices.length === 0 ? (
   589→                  <SelectItem value="none" disabled>No invoices for this customer</SelectItem>
   590→                ) : (
   591→                  custInvoices.map((i) => (
   592→                    <SelectItem key={i.id} value={i.id}>
   593→                      <span className="font-money">{i.invoiceNumber}</span> · {formatDate(i.invoiceDate)} · {formatINR(Number(i.grandTotal))}
   594→                    </SelectItem>
   595→                  ))
   596→                )}
   597→              </SelectContent>
   598→            </Select>
   599→          </Field>
   600→          <Field label="Return date">
   601→            <Input type="date" value={returnDate} onChange={(e) => setReturnDate(e.target.value)} className={inputCls} />
   602→          </Field>
   603→        </div>
   604→
   605→        {/* Item rows */}
   606→        <div className="space-y-2">
   607→          <div className="flex items-center justify-between">
   608→            <span className="text-[11px] uppercase tracking-wider font-semibold text-dmk-text-muted">Returned items</span>
   609→            <Button size="sm" variant="outline" className="h-8 border-dmk-border-subtle text-dmk-text-secondary hover:bg-dmk-hover" onClick={addItem} disabled={products.length === 0}>
   610→              <Plus className="h-3.5 w-3.5" /> Add item
   611→            </Button>
   612→          </div>
   613→          {items.length === 0 ? (
   614→            <div className="dmk-well p-4 text-center text-[12px] text-dmk-text-muted">No item rows — click “Add item”.</div>
   615→          ) : (
   616→            <div className="space-y-2 max-h-64 overflow-y-auto pr-1">
   617→              {items.map((it, idx) => {
   618→                const p = products.find((x) => x.id === it.productId);
   619→                const lineTaxable = round2(Number(it.unitPrice) * it.qty);
   620→                const g = p ? previewGst(lineTaxable, p.gstRate, firmStateCode, customers.find((c) => c.id === customerId)?.stateCode ?? firmStateCode) : { cgst: 0, sgst: 0, igst: 0 };
   621→                return (
   622→                  <div key={idx} className="dmk-well p-2.5 grid grid-cols-12 gap-2 items-center">
   623→                    <div className="col-span-12 sm:col-span-5">
   624→                      <Select
   625→                        value={it.productId}
   626→                        onValueChange={(v) => {
   627→                          const np = products.find((x) => x.id === v);
   628→                          updateItem(idx, { productId: v, unitPrice: np ? Number(np.tier4Retailer) || 0 : it.unitPrice });
   629→                        }}
   630→                      >
   631→                        <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   632→                        <SelectContent className="max-h-64">
   633→                          {products.map((pr) => (
   634→                            <SelectItem key={pr.id} value={pr.id}>
   635→                              <span className="font-money">{pr.sku}</span> · {pr.name}
   636→                            </SelectItem>
   637→                          ))}
   638→                        </SelectContent>
   639→                      </Select>
   640→                    </div>
   641→                    <div className="col-span-4 sm:col-span-2">
   642→                      <Input
   643→                        type="number"
   644→                        min={1}
   645→                        step={1}
   646→                        value={it.qty}
   647→                        onChange={(e) => updateItem(idx, { qty: Math.max(1, Number(e.target.value) || 1) })}
   648→                        aria-label="Returned quantity"
   649→                        className={cn(inputCls, "font-money")}
   650→                        placeholder="Qty"
   651→                      />
   652→                    </div>
   653→                    <div className="col-span-8 sm:col-span-3">
   654→                      <Select value={it.defectType} onValueChange={(v) => updateItem(idx, { defectType: v as DraftItem["defectType"] })}>
   655→                        <SelectTrigger className={cn(inputCls, "w-full")}><SelectValue /></SelectTrigger>
   656→                        <SelectContent>
   657→                          {DEFECTS.map((d) => (
   658→                            <SelectItem key={d} value={d}>{d}</SelectItem>
   659→                          ))}
   660→                        </SelectContent>
   661→                      </Select>
   662→                    </div>
   663→                    <div className="col-span-10 sm:col-span-1.5">
   664→                      <Input
   665→                        type="number"
   666→                        min={0}
   667→                        step={0.01}
   668→                        value={it.unitPrice}
   669→                        onChange={(e) => updateItem(idx, { unitPrice: Number(e.target.value) || 0 })}
   670→                        aria-label="Unit price"
   671→                        className={cn(inputCls, "font-money")}
   672→                        placeholder="₹"
   673→                      />
   674→                    </div>
   675→                    <div className="col-span-2 sm:col-span-0.5 flex justify-end">
   676→                      <button
   677→                        onClick={() => setItems((prev) => prev.filter((_, i) => i !== idx))}
   678→                        aria-label="Remove item"
   679→                        className="h-8 w-8 inline-flex items-center justify-center rounded-md text-dmk-text-muted hover:text-dmk-danger hover:bg-dmk-hover transition-colors"
   680→                      >
   681→                        <Trash2 className="h-4 w-4" />
   682→                      </button>
   683→                    </div>
   684→                    <div className="col-span-12 flex items-center justify-between text-[11px] text-dmk-text-muted px-1">
   685→                      <span>
   686→                        Taxable <span className="font-money text-dmk-text-secondary">{formatINR(lineTaxable)}</span>
   687→                        {" · "}GST {p?.gstRate ?? 0}% <span className="font-money text-dmk-text-secondary">{formatINR(g.cgst + g.sgst + g.igst)}</span>
   688→                      </span>
   689→                      <span>
   690→                        Line total <span className="font-money text-dmk-text-primary">{formatINR(round2(lineTaxable + g.cgst + g.sgst + g.igst))}</span>
   691→                      </span>
   692→                    </div>
   693→                  </div>
   694→                );
   695→              })}
   696→            </div>
   697→          )}
   698→        </div>
   699→
   700→        <div className="flex items-center justify-between dmk-well px-3 py-2.5">
   701→          <div className="flex items-center gap-1.5 text-[11.5px] text-dmk-warning">
   702→            <AlertTriangle className="h-3.5 w-3.5" /> Qty quarantined to Damaged Stock
   703→          </div>
   704→          <div className="text-right">
   705→            <span className="text-[11px] text-dmk-text-muted block">Preview total (taxable {formatINR(draftTotals.taxable)} + tax {formatINR(draftTotals.tax)})</span>
   706→            <span className="font-money text-[16px] text-dmk-yellow">{formatINR(draftTotals.grand)}</span>
   707→          </div>
   708→        </div>
   709→
   710→        <DialogFooter>
   711→          <Button variant="outline" onClick={() => onOpenChange(false)} className="border-dmk-border-medium text-dmk-text-secondary hover:bg-dmk-hover">Cancel</Button>
   712→          <Button onClick={submit} disabled={saving || items.length === 0} className="bg-dmk-yellow text-white hover:bg-dmk-yellow/90">
   713→            {saving ? "Posting…" : "Create credit note"}
   714→          </Button>
   715→        </DialogFooter>
   716→      </DialogContent>
   717→    </Dialog>
   718→  );
   719→}
